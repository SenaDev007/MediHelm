   130→function formatDate(dateStr: string) {
   131→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   132→    day: '2-digit',
   133→    month: '2-digit',
   134→    year: 'numeric',
   135→  })

   138→function daysUntil(dateStr: string): number {
   139→  const now = new Date()
   140→  const target = new Date(dateStr)
   141→  const diff = target.getTime() - now.getTime()
   142→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
   143→}

   145→function getStockTotal(medicament: Medicament): number {
   146→  return medicament.lots.reduce((sum, lot) => sum + lot.quantite, 0)
   147→}
   148→
   149→type StockStatus = 'OK' | 'ALERTE' | 'RUPTURE'
   150→
   151→function getStockStatus(medicament: Medicament): StockStatus {
   152→  const total = getStockTotal(medicament)
   153→  if (total === 0) return 'RUPTURE'
   154→  if (total <= medicament.stockMinimum) return 'ALERTE'
   155→  return 'OK'
   156→}

   158→function getStatusBadge(status: StockStatus) {
   159→  switch (status) {
   160→    case 'OK':
   161→      return <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-0 text-xs">OK</Badge>
   162→    case 'ALERTE':
   163→      return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100 border-0 text-xs">Alerte</Badge>
   164→    case 'RUPTURE':
   165→      return <Badge className="bg-red-100 text-red-700 hover:bg-red-100 border-0 text-xs">Rupture</Badge>
   166→  }
   167→}
   168→
   169→function getDaysRemainingBadge(days: number) {
   170→  if (days <= 0) {
   171→    return <Badge variant="destructive" className="text-xs">Expiré</Badge>
   172→  }
   173→  if (days < 30) {
   174→    return <Badge variant="destructive" className="text-xs">{days}j</Badge>
   175→  }
   176→  if (days < 60) {
   177→    return <Badge className="bg-orange-100 text-orange-700 hover:bg-orange-100 border-0 text-xs">{days}j</Badge>
   178→  }
   179→  if (days <= 90) {
   180→    return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100 border-0 text-xs">{days}j</Badge>
   181→  }
   182→  return <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-0 text-xs">{days}j</Badge>
   183→}
   184→
   185→function getAlertTypeBadge(type: string) {
   186→  switch (type) {
   187→    case 'RUPTURE':
   188→      return <Badge variant="destructive" className="text-xs">Rupture</Badge>
   189→    case 'SEUIL_MINIMUM':
   190→      return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100 border-0 text-xs">Seuil min.</Badge>
   191→    case 'PEREMPTION_PROCHE':
   192→      return <Badge className="bg-orange-100 text-orange-700 hover:bg-orange-100 border-0 text-xs">Péremption</Badge>
   193→    case 'SURSTOCK':
   194→      return <Badge className="bg-blue-100 text-blue-700 hover:bg-blue-100 border-0 text-xs">Surstock</Badge>
   195→    default:
   196→      return <Badge variant="outline" className="text-xs">{type}</Badge>
   197→  }
   198→}
