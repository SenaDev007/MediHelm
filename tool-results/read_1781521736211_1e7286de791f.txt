   134→export default function CaissePage() {
   135→  const { user, pharmacie, isAuthenticated } = useAuth()
   136→  const pharmacieId = pharmacie?.id || ''
   137→
   138→  // === Session Caisse State ===
   139→  const [caisses, setCaisses] = useState<CaisseData[]>([])
   140→  const [activeSession, setActiveSession] = useState<SessionCaisseData | null>(null)
   141→  const [loadingSession, setLoadingSession] = useState(true)
   142→  const [openSessionDialog, setOpenSessionDialog] = useState(false)
   143→  const [selectedCaisseId, setSelectedCaisseId] = useState('')
   144→  const [fondDeCaisse, setFondDeCaisse] = useState('10000')
   145→
   146→  // === Product Search State ===
   147→  const [searchQuery, setSearchQuery] = useState('')
   148→  const [searchResults, setSearchResults] = useState<MedicamentResult[]>([])
   149→  const [searching, setSearching] = useState(false)
   150→  const searchTimeoutRef = useRef<NodeJS.Timeout | null>(null)
   151→
   152→  // === Cart State ===
   153→  const [cart, setCart] = useState<CartItem[]>([])
   154→  const [globalRemise, setGlobalRemise] = useState(0)
   155→
   156→  // === Patient/Ordonnance Linking ===
   157→  const [patientSearch, setPatientSearch] = useState('')
   158→  const [patientResults, setPatientResults] = useState<PatientResult[]>([])
   159→  const [selectedPatient, setSelectedPatient] = useState<PatientResult | null>(null)
   160→  const [ordonnances, setOrdonnances] = useState<OrdonnanceResult[]>([])
   161→  const [selectedOrdonnance, setSelectedOrdonnance] = useState<OrdonnanceResult | null>(null)
   162→
   163→  // === Payment State ===
   164→  const [paymentDialog, setPaymentDialog] = useState(false)
   165→  const [paiements, setPaiements] = useState<{ mode: ModePaiement; montant: string }[]>([
   166→    { mode: 'ESPECES', montant: '' },
   167→  ])
   168→  const [processingSale, setProcessingSale] = useState(false)
   169→
   170→  // === Sale Result State ===
   171→  const [saleResult, setSaleResult] = useState<{ success: boolean; venteId?: string; error?: string } | null>(null)
   172→  const [venteType, setVenteType] = useState<'COMPTOIR' | 'ORDONNANCE'>('COMPTOIR')
   173→
   174→  // === Close Session State ===
   175→  const [closeSessionDialog, setCloseSessionDialog] = useState(false)
   176→  const [soldePhysique, setSoldePhysique] = useState('')
   177→  const [closingSession, setClosingSession] = useState(false)
   178→
   179→  // === Fetch Caisses and Active Session ===
   180→  const fetchCaissesAndSession = useCallback(async () => {
   181→    if (!pharmacieId) return
   182→    setLoadingSession(true)
   183→    try {
   184→      const [caissesRes, sessionsRes] = await Promise.all([
   185→        fetch(`/api/caisses?pharmacieId=${pharmacieId}`),
   186→        fetch(`/api/sessions-caisse?pharmacieId=${pharmacieId}&statut=ouverte`),
   187→      ])
   188→      if (caissesRes.ok) setCaisses(await caissesRes.json())
   189→      if (sessionsRes.ok) {
   190→        const sessions: SessionCaisseData[] = await sessionsRes.json()
   191→        const mySession = sessions.find((s: SessionCaisseData) => s.utilisateur.id === user?.id)
   192→        setActiveSession(mySession || sessions[0] || null)
   193→      }
   194→    } catch (err) {
   195→      console.error('Erreur chargement caisses:', err)
   196→    } finally {
   197→      setLoadingSession(false)
   198→    }
   199→  }, [pharmacieId, user?.id])
   200→
   201→  useEffect(() => {
   202→    fetchCaissesAndSession()
   203→  }, [fetchCaissesAndSession])
   204→
   205→  // === Fetch Ordonnances ===
   206→  useEffect(() => {
   207→    if (!pharmacieId) return
   208→    fetch(`/api/ordonnances?pharmacieId=${pharmacieId}&statut=VALIDEE`)
   209→      .then((r) => r.json())
   210→      .then(setOrdonnances)
   211→      .catch(() => {})
   212→  }, [pharmacieId])
   213→
   214→  // === Product Search (debounced) ===
   215→  const searchProducts = useCallback(
   216→    async (query: string) => {
   217→      if (!pharmacieId || query.length < 2) {
   218→        setSearchResults([])
   219→        return
   220→      }
   221→      setSearching(true)
   222→      try {
   223→        const res = await fetch(`/api/medicaments?pharmacieId=${pharmacieId}&search=${encodeURIComponent(query)}`)
   224→        if (res.ok) setSearchResults(await res.json())
   225→      } catch {
   226→        setSearchResults([])
   227→      } finally {
   228→        setSearching(false)
   229→      }
   230→    },
   231→    [pharmacieId]
   232→  )
   233→
   234→  useEffect(() => {
   235→    if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current)
   236→    searchTimeoutRef.current = setTimeout(() => searchProducts(searchQuery), 300)
   237→    return () => {
   238→      if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current)
   239→    }
   240→  }, [searchQuery, searchProducts])
   241→
   242→  // === Patient Search (debounced) ===
   243→  const searchPatients = useCallback(
   244→    async (query: string) => {
   245→      if (!pharmacieId || query.length < 2) {
   246→        setPatientResults([])
   247→        return
   248→      }
   249→      try {
   250→        const res = await fetch(`/api/patients?pharmacieId=${pharmacieId}&search=${encodeURIComponent(query)}`)
   251→        if (res.ok) setPatientResults(await res.json())
   252→      } catch {
   253→        setPatientResults([])
   254→      }
   255→    },
   256→    [pharmacieId]
   257→  )
   258→
   259→  useEffect(() => {
   260→    const timeout = setTimeout(() => searchPatients(patientSearch), 300)
   261→    return () => clearTimeout(timeout)
   262→  }, [patientSearch, searchPatients])
   263→
   264→  // === Cart Operations ===
   265→  const addToCart = (med: MedicamentResult) => {
   266→    const totalStock = med.lots.reduce((s, l) => s + l.quantite, 0)
   267→    setCart((prev) => {
   268→      const existing = prev.find((i) => i.medicamentId === med.id)
   269→      if (existing) {
   270→        if (existing.quantite >= totalStock) return prev
   271→        return prev.map((i) =>
   272→          i.medicamentId === med.id ? { ...i, quantite: i.quantite + 1 } : i
   273→        )
   274→      }
   275→      const firstLot = med.lots[0]
   276→      return [
   277→        ...prev,
   278→        {
   279→          medicamentId: med.id,
   280→          nomCommercial: med.nomCommercial,
   281→          dci: med.dci,
   282→          prixUnitaire: med.prixVente,
   283→          quantite: 1,
   284→          stock: totalStock,
   285→          remise: 0,
   286→          lotId: firstLot?.id,
   287→        },
   288→      ]
   289→    })
   290→    setSearchQuery('')
   291→    setSearchResults([])
   292→  }
   293→
   294→  const updateQty = (medicamentId: string, delta: number) => {
   295→    setCart((prev) =>
   296→      prev.map((i) => {
   297→        if (i.medicamentId !== medicamentId) return i
   298→        const newQty = i.quantite + delta
   299→        if (newQty <= 0) return i
   300→        if (newQty > i.stock) return i
   301→        return { ...i, quantite: newQty }
   302→      })
   303→    )
   304→  }
   305→
   306→  const removeFromCart = (medicamentId: string) => {
   307→    setCart((prev) => prev.filter((i) => i.medicamentId !== medicamentId))
   308→  }
   309→
   310→  const updateItemRemise = (medicamentId: string, remise: number) => {
   311→    setCart((prev) =>
   312→      prev.map((i) =>
   313→        i.medicamentId === medicamentId
   314→          ? { ...i, remise: Math.min(Math.max(remise, 0), i.prixUnitaire * i.quantite) }
   315→          : i
   316→      )
   317→    )
   318→  }
   319→
   320→  // === Totals ===
   321→  const sousTotal = cart.reduce((s, i) => s + i.prixUnitaire * i.quantite - i.remise, 0)
   322→  const montantRemise = globalRemise
   323→  const montantTotal = Math.max(sousTotal - montantRemise, 0)
   324→
   325→  // === Open Session ===
   326→  const handleOpenSession = async () => {
   327→    if (!selectedCaisseId || !user?.id) return
   328→    try {
   329→      const res = await fetch('/api/sessions-caisse', {
   330→        method: 'POST',
   331→        headers: { 'Content-Type': 'application/json' },
   332→        body: JSON.stringify({
   333→          pharmacieId,
   334→          utilisateurId: user.id,
   335→          caisseId: selectedCaisseId,
   336→          fondDeCaisse: parseFloat(fondDeCaisse) || 0,
   337→        }),
   338→      })
   339→      if (res.ok) {
   340→        const session = await res.json()
   341→        setActiveSession(session)
   342→        setOpenSessionDialog(false)
   343→      }
   344→    } catch (err) {
   345→      console.error('Erreur ouverture session:', err)
   346→    }
   347→  }
   348→
   349→  // === Close Session ===
   350→  const handleCloseSession = async () => {
   351→    if (!activeSession) return
   352→    setClosingSession(true)
   353→    try {
   354→      const res = await fetch(`/api/sessions-caisse/${activeSession.id}`, {
   355→        method: 'PATCH',
   356→        headers: { 'Content-Type': 'application/json' },
   357→        body: JSON.stringify({
   358→          soldePhysique: parseFloat(soldePhysique) || 0,
   359→        }),
   360→      })
   361→      if (res.ok) {
   362→        setActiveSession(null)
   363→        setCloseSessionDialog(false)
   364→        setSoldePhysique('')
   365→        fetchCaissesAndSession()
   366→      }
   367→    } catch (err) {
   368→      console.error('Erreur fermeture session:', err)
   369→    } finally {
   370→      setClosingSession(false)
   371→    }
   372→  }
   373→
   374→  // === Payment ===
   375→  const totalPaiements = paiements.reduce((s, p) => s + (parseFloat(p.montant) || 0), 0)
   376→  const resteAPayer = montantTotal - totalPaiements
   377→
   378→  const addPaiement = () => {
   379→    setPaiements([...paiements, { mode: 'ESPECES', montant: '' }])
   380→  }
   381→
   382→  const removePaiement = (idx: number) => {
   383→    if (paiements.length <= 1) return
   384→    setPaiements(paiements.filter((_, i) => i !== idx))
   385→  }
   386→
   387→  const updatePaiement = (idx: number, field: 'mode' | 'montant', value: string) => {
   388→    setPaiements(
   389→      paiements.map((p, i) => (i === idx ? { ...p, [field]: value } : p))
   390→    )
   391→  }
   392→
   393→  // === Create Vente ===
   394→  const handleCreateVente = async () => {
   395→    if (!activeSession || !user?.id || cart.length === 0) return
   396→    if (resteAPayer > 0.5) return // Must pay full amount
   397→
   398→    setProcessingSale(true)
   399→    try {
   400→      const lignes = cart.map((item) => ({
   401→        medicamentId: item.medicamentId,
   402→        lotId: item.lotId || undefined,
   403→        quantite: item.quantite,
   404→        prixUnitaire: item.prixUnitaire,
   405→        montant: item.prixUnitaire * item.quantite - item.remise,
   406→        remise: item.remise,
   407→      }))
   408→
   409→      const paiementData = paiements
   410→        .filter((p) => parseFloat(p.montant) > 0)
   411→        .map((p) => ({
   412→          modePaiement: p.mode,
   413→          montant: parseFloat(p.montant),
   414→        }))
   415→
   416→      const body = {
   417→        pharmacieId,
   418→        utilisateurId: user.id,
   419→        sessionId: activeSession.id,
   420→        patientId: selectedPatient?.id || undefined,
   421→        ordonnanceId: selectedOrdonnance?.id || undefined,
   422→        typeVente: venteType,
   423→        statut: 'VALIDEE',
   424→        montantTotal,
   425→        montantRemise,
   426→        montantPaye: totalPaiements,
   427→        monnaieRendue: Math.max(totalPaiements - montantTotal, 0),
   428→        lignes,
   429→        paiements: paiementData,
   430→      }
   431→
   432→      const res = await fetch('/api/ventes', {
   433→        method: 'POST',
   434→        headers: { 'Content-Type': 'application/json' },
   435→        body: JSON.stringify(body),
   436→      })
   437→
   438→      if (res.ok) {
   439→        const vente = await res.json()
   440→        setSaleResult({ success: true, venteId: vente.id })
   441→        setCart([])
   442→        setGlobalRemise(0)
   443→        setSelectedPatient(null)
   444→        setSelectedOrdonnance(null)
   445→        setPatientSearch('')
   446→        setPaiements([{ mode: 'ESPECES', montant: '' }])
   447→        setPaymentDialog(false)
   448→        fetchCaissesAndSession()
   449→      } else {
   450→        const err = await res.json()
   451→        setSaleResult({ success: false, error: err.error || 'Erreur lors de la vente' })
   452→      }
   453→    } catch {
   454→      setSaleResult({ success: false, error: 'Erreur réseau' })
   455→    } finally {
   456→      setProcessingSale(false)
   457→    }
   458→  }
   459→
   460→  // === Print Ticket ===
   461→  const handlePrintTicket = (venteId: string) => {
   462→    window.open(`/api/ticket?type=ticket&venteId=${venteId}`, '_blank')
   463→  }
   464→
   465→  // === Quick Actions ===
   466→  const startVenteRapide = () => {
   467→    setVenteType('COMPTOIR')
   468→    setSelectedPatient(null)
   469→    setSelectedOrdonnance(null)
   470→    setPatientSearch('')
   471→  }
   472→
   473→  const startVenteOrdonnance = () => {
   474→    setVenteType('ORDONNANCE')
   475→  }
   476→
   477→  // === Loading / Unauthenticated ===
   478→  if (!isAuthenticated) {
   479→    return (
   480→      <div className="flex h-full items-center justify-center">
   481→        <p className="text-muted-foreground">Connectez-vous pour accéder à la caisse</p>
   482→      </div>
   483→    )
   484→  }
   485→
   486→  return (
   487→    <div className="flex flex-col h-full -m-4 md:-m-6">
   488→      {/* === Session Caisse Banner === */}
   489→      <div
   490→        className="border-b px-4 py-2 flex items-center justify-between flex-wrap gap-2"
   491→        style={{ background: activeSession ? '#E1F5EE' : '#FEF3C7' }}
   492→      >
   493→        <div className="flex items-center gap-3">
   494→          {activeSession ? (
   495→            <>
   496→              <Unlock className="h-4 w-4 text-[#1D9E75]" />
   497→              <span className="text-sm font-medium">
   498→                Session ouverte — <strong>{activeSession.caisse.nom}</strong>
   499→              </span>
   500→              <Badge variant="outline" className="text-[#085041] border-[#1D9E75]">
   501→                Fond : {fcfa(activeSession.fondDeCaisse)}
   502→              </Badge>
   503→              <span className="text-xs text-muted-foreground">
   504→                par {activeSession.utilisateur.prenom} {activeSession.utilisateur.nom}
   505→              </span>
   506→            </>
   507→          ) : (
   508→            <>
   509→              <Lock className="h-4 w-4 text-amber-600" />
   510→              <span className="text-sm font-medium text-amber-800">
   511→                Aucune session ouverte — ouvrez une session pour encaisser
   512→              </span>
   513→            </>
   514→          )}
   515→        </div>
   516→        <div className="flex gap-2">
   517→          {activeSession ? (
   518→            <Button
   519→              size="sm"
   520→              variant="outline"
   521→              className="border-red-300 text-red-700 hover:bg-red-50"
   522→              onClick={() => setCloseSessionDialog(true)}
   523→            >
   524→              Fermer la session
   525→            </Button>
   526→          ) : (
   527→            <Button
   528→              size="sm"
   529→              style={{ background: '#1D9E75' }}
   530→              className="text-white hover:opacity-90"
   531→              onClick={() => setOpenSessionDialog(true)}
   532→            >
   533→              Ouvrir une session
   534→            </Button>
   535→          )}
   536→        </div>
   537→      </div>
   538→
   539→      {/* === Quick Actions === */}
   540→      <div className="px-4 py-2 border-b bg-background flex items-center gap-2 flex-wrap">
   541→        <Button
   542→          size="sm"
   543→          variant={venteType === 'COMPTOIR' ? 'default' : 'outline'}
   544→          style={venteType === 'COMPTOIR' ? { background: '#1D9E75' } : {}}
   545→          onClick={startVenteRapide}
   546→          className="text-sm"
   547→        >
   548→          <ShoppingCart className="h-3.5 w-3.5 mr-1.5" />
   549→          Vente rapide
   550→        </Button>
   551→        <Button
   552→          size="sm"
   553→          variant={venteType === 'ORDONNANCE' ? 'default' : 'outline'}
   554→          style={venteType === 'ORDONNANCE' ? { background: '#1D9E75' } : {}}
   555→          onClick={startVenteOrdonnance}
   556→          className="text-sm"
   557→        >
   558→          <FileText className="h-3.5 w-3.5 mr-1.5" />
   559→          Vente ordonnance
   560→        </Button>
   561→
   562→        {/* Patient linking */}
   563→        <div className="ml-auto flex items-center gap-2">
   564→          {selectedPatient ? (
   565→            <Badge variant="secondary" className="gap-1 pr-1">
   566→              <UserPlus className="h-3 w-3" />
   567→              {selectedPatient.prenom} {selectedPatient.nom}
   568→              <button
   569→                className="ml-1 rounded-full hover:bg-muted p-0.5"
   570→                onClick={() => setSelectedPatient(null)}
   571→              >
   572→                <XCircle className="h-3 w-3" />
   573→              </button>
   574→            </Badge>
   575→          ) : (
   576→            <div className="relative">
