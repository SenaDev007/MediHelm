    61→  caisseId: string

    65→  soldeCloture: number | null

    68→  fermeLe: string | null

    70→  utilisateur?: { id: string; nom: string; prenom: string }
    71→  ventes?: VenteSimple[]

   100→  prenom: string

   115→

   181→
   182→export default function CaissePage() {
   183→  const { pharmacie, user } = useAuth()
   184→  const pharmacieId = pharmacie?.id || ''
   185→  const utilisateurId = user?.id || ''
   186→
   187→  // Session state
   188→  const [activeSession, setActiveSession] = useState<SessionCaisse | null>(null)
   189→  const [caisses, setCaisses] = useState<Caisse[]>([])
   190→  const [sessionLoading, setSessionLoading] = useState(true)
   191→
   192→  // Open register dialog
   193→  const [openDialogOpen, setOpenDialogOpen] = useState(false)
   194→  const [selectedCaisseId, setSelectedCaisseId] = useState('')
   195→  const [soldeOuverture, setSoldeOuverture] = useState('')
   196→  const [openingSession, setOpeningSession] = useState(false)
   197→
   198→  // Close register dialog
   199→  const [closeDialogOpen, setCloseDialogOpen] = useState(false)
   200→  const [soldeCloture, setSoldeCloture] = useState('')
   201→  const [closingSession, setClosingSession] = useState(false)
   202→
   203→  // POS state
   204→  const [cart, setCart] = useState<CartItem[]>([])
   205→  const [medSearch, setMedSearch] = useState('')
   206→  const [medicaments, setMedicaments] = useState<Medicament[]>([])
   207→  const [medSearchLoading, setMedSearchLoading] = useState(false)
   208→  const [selectedPatientId, setSelectedPatientId] = useState<string>('')
   209→  const [patients, setPatients] = useState<Patient[]>([])
   210→  const [patientSearch, setPatientSearch] = useState('')
   211→  const [showNewPatient, setShowNewPatient] = useState(false)
   212→  const [newPatient, setNewPatient] = useState({ nom: '', prenom: '', telephone: '' })
   213→  const [posGlobalRemise, setPosGlobalRemise] = useState(0)
   214→
   215→  // Payment state
   216→  const [splitPaiements, setSplitPaiements] = useState<SplitPaiement[]>([
   217→    { id: '1', mode: 'ESPECES', montant: 0, reference: '' },
   218→  ])
   219→  const [showSplitPayment, setShowSplitPayment] = useState(false)
   220→  const [submitting, setSubmitting] = useState(false)
   221→
   222→  // Recent transactions
   223→  const [recentVentes, setRecentVentes] = useState<VenteSimple[]>([])
   224→
   225→  // ============================================================
   226→  // Fetch active session & caisses
   227→  // ============================================================
   228→
   229→  const fetchSessionData = useCallback(async () => {
   230→    if (!pharmacieId) return
   231→    setSessionLoading(true)
   232→    try {
   233→      const [sessionRes, caissesRes] = await Promise.all([
   234→        fetch(`/api/sessions-caisse?pharmacieId=${pharmacieId}&statut=OUVERTE`),
   235→        fetch(`/api/caisses?pharmacieId=${pharmacieId}`),
   236→      ])
   237→
   238→      if (sessionRes.ok) {
   239→        const sessions: SessionCaisse[] = await sessionRes.json()
   240→        if (sessions.length > 0) {
   241→          setActiveSession(sessions[0])
   242→          setRecentVentes(sessions[0].ventes || [])
   243→        } else {
   244→          setActiveSession(null)
   245→          setRecentVentes([])
   246→        }
   247→      }
   248→
   249→      if (caissesRes.ok) {
   250→        setCaisses(await caissesRes.json())
   251→      }
   252→    } catch {
   253→      toast.error('Erreur lors du chargement de la caisse')
   254→    } finally {
   255→      setSessionLoading(false)
   256→    }
   257→  }, [pharmacieId])
   258→
   259→  useEffect(() => {
   260→    fetchSessionData()
   261→  }, [fetchSessionData])
   262→
   263→  // ============================================================
   264→  // Fetch medicaments for POS search
   265→  // ============================================================
   266→
   267→  const fetchMedicaments = useCallback(async () => {
   268→    if (!pharmacieId || !medSearch) {
   269→      setMedicaments([])
   270→      return
   271→    }
   272→    setMedSearchLoading(true)
   273→    try {
   274→      const res = await fetch(`/api/medicaments?pharmacieId=${pharmacieId}&search=${encodeURIComponent(medSearch)}&pageSize=10`)
   275→      if (res.ok) {
   276→        const data = await res.json()
   277→        setMedicaments(data.data || data || [])
   278→      }
   279→    } catch {
   280→      // silent
   281→    } finally {
   282→      setMedSearchLoading(false)
   283→    }
   284→  }, [pharmacieId, medSearch])
   285→
   286→  useEffect(() => {
   287→    const timeout = setTimeout(fetchMedicaments, 300)
   288→    return () => clearTimeout(timeout)
   289→  }, [fetchMedicaments])
   290→
   291→  // ============================================================
   292→  // Fetch patients for POS
   293→  // ============================================================
   294→
   295→  const fetchPatients = useCallback(async () => {
   296→    if (!pharmacieId) return
   297→    try {
   298→      const params = new URLSearchParams({ pharmacieId })
   299→      if (patientSearch) params.set('search', patientSearch)
   300→      const res = await fetch(`/api/patients?${params}`)
   301→      if (res.ok) setPatients(await res.json())
   302→    } catch {
   303→      // silent
   304→    }
   305→  }, [pharmacieId, patientSearch])
   306→
   307→  useEffect(() => {
   308→    const timeout = setTimeout(fetchPatients, 300)
   309→    return () => clearTimeout(timeout)
   310→  }, [fetchPatients])
   311→
   312→  // ============================================================
   313→  // Cart calculations
   314→  // ============================================================
   315→
   316→  const cartSubtotal = useMemo(() => {
   317→    return cart.reduce((sum, item) => sum + (item.prixUnitaire * item.quantite - item.remise), 0)
   318→  }, [cart])
   319→
   320→  const cartTotal = useMemo(() => {
   321→    return Math.max(0, cartSubtotal - posGlobalRemise)
   322→  }, [cartSubtotal, posGlobalRemise])
   323→
   324→  const totalPaidAmount = useMemo(() => {
   325→    return splitPaiements.reduce((sum, p) => sum + p.montant, 0)
   326→  }, [splitPaiements])
   327→
   328→  const remainingAmount = useMemo(() => {
   329→    return Math.max(0, cartTotal - totalPaidAmount)
   330→  }, [cartTotal, totalPaidAmount])
   331→
   332→  // ============================================================
   333→  // Cart actions
   334→  // ============================================================
   335→
   336→  const addToCart = (med: Medicament) => {
   337→    setCart(prev => {
   338→      const existing = prev.find(i => i.medicamentId === med.id)
   339→      if (existing) {
   340→        return prev.map(i =>
   341→          i.medicamentId === med.id
   342→            ? { ...i, quantite: i.quantite + 1 }
   343→            : i
   344→        )
   345→      }
   346→      return [...prev, {
   347→        medicamentId: med.id,
   348→        nomCommercial: med.nomCommercial,
   349→        dci: med.dci,
   350→        prixUnitaire: med.prixPublic,
   351→        quantite: 1,
   352→        remise: 0,
   353→      }]
   354→    })
   355→    toast.success(`${med.nomCommercial} ajouté`)
   356→  }
   357→
   358→  const removeFromCart = (medicamentId: string) => {
   359→    setCart(prev => prev.filter(i => i.medicamentId !== medicamentId))
   360→  }
   361→
   362→  const updateCartItem = (medicamentId: string, field: 'quantite' | 'remise', value: number) => {
   363→    setCart(prev => prev.map(i =>
   364→      i.medicamentId === medicamentId
   365→        ? { ...i, [field]: Math.max(0, value) }
   366→        : i
   367→    ))
   368→  }
   369→
   370→  const clearCart = () => {
   371→    setCart([])
   372→    setPosGlobalRemise(0)
   373→    setSelectedPatientId('')
   374→    setSplitPaiements([{ id: '1', mode: 'ESPECES', montant: 0, reference: '' }])
   375→    setShowSplitPayment(false)
   376→  }
   377→
   378→  // ============================================================
   379→  // Open register
   380→  // ============================================================
   381→
   382→  const handleOpenRegister = async () => {
   383→    if (!selectedCaisseId) {
   384→      toast.error('Sélectionnez une caisse')
   385→      return
   386→    }
   387→    setOpeningSession(true)
   388→    try {
   389→      const res = await fetch('/api/sessions-caisse', {
   390→        method: 'POST',
   391→        headers: { 'Content-Type': 'application/json' },
   392→        body: JSON.stringify({
   393→          pharmacieId,
   394→          caisseId: selectedCaisseId,
   395→          utilisateurId,
   396→          soldeOuverture: parseFloat(soldeOuverture) || 0,
   397→        }),
   398→      })
   399→      if (!res.ok) {
   400→        const err = await res.json()
   401→        throw new Error(err.error || 'Erreur')
   402→      }
   403→      const session = await res.json()
   404→      setActiveSession(session)
   405→      setOpenDialogOpen(false)
   406→      setSelectedCaisseId('')
   407→      setSoldeOuverture('')
   408→      toast.success('Caisse ouverte avec succès')
   409→    } catch (err) {
   410→      toast.error(err instanceof Error ? err.message : 'Erreur lors de l\'ouverture')
   411→    } finally {
   412→      setOpeningSession(false)
   413→    }
   414→  }
   415→
   416→  // ============================================================
   417→  // Close register
   418→  // ============================================================
   419→
   420→  const handleCloseRegister = async () => {
   421→    if (!activeSession) return
   422→    setClosingSession(true)
   423→    try {
   424→      const res = await fetch(`/api/sessions-caisse/${activeSession.id}`, {
   425→        method: 'PATCH',
   426→        headers: { 'Content-Type': 'application/json' },
   427→        body: JSON.stringify({
   428→          action: 'cloturer',
   429→          soldeCloture: parseFloat(soldeCloture) || 0,
   430→        }),
   431→      })
   432→      if (!res.ok) {
   433→        const err = await res.json()
   434→        throw new Error(err.error || 'Erreur')
   435→      }
   436→      toast.success('Caisse clôturée avec succès')
   437→      setCloseDialogOpen(false)
   438→      setSoldeCloture('')
   439→      setActiveSession(null)
   440→      setRecentVentes([])
   441→      clearCart()
   442→    } catch (err) {
   443→      toast.error(err instanceof Error ? err.message : 'Erreur lors de la clôture')
   444→    } finally {
   445→      setClosingSession(false)
   446→    }
   447→  }
   448→
   449→  // ============================================================
   450→  // Submit sale (POS)
   451→  // ============================================================
   452→
   453→  const handleSubmitSale = async () => {
   454→    if (cart.length === 0) {
   455→      toast.error('Ajoutez au moins un médicament')
   456→      return
   457→    }
   458→    if (remainingAmount > 0 && !showSplitPayment) {
   459→      toast.error('Le montant payé est insuffisant')
   460→      return
   461→    }
   462→    if (showSplitPayment && remainingAmount > 0) {
   463→      toast.error(`Il reste ${formatFCFA(remainingAmount)} à payer`)
   464→      return
   465→    }
   466→
   467→    setSubmitting(true)
   468→    try {
   469→      // Build payments
   470→      const paiementData = showSplitPayment
   471→        ? splitPaiements.filter(p => p.montant > 0).map(p => ({
   472→          montant: p.montant,
   473→          mode: p.mode,
   474→          reference: p.reference || null,
   475→        }))
   476→        : [{ montant: cartTotal, mode: splitPaiements[0]?.mode || 'ESPECES' as ModePaiement }]
   477→
   478→      const res = await fetch('/api/ventes', {
   479→        method: 'POST',
   480→        headers: { 'Content-Type': 'application/json' },
   481→        body: JSON.stringify({
   482→          pharmacieId,
   483→          patientId: selectedPatientId || null,
   484→          sessionId: activeSession?.id || null,
   485→          utilisateurId,
   486→          remise: posGlobalRemise,
   487→          lignes: cart.map(item => ({
   488→            medicamentId: item.medicamentId,
   489→            quantite: item.quantite,
   490→            prixUnitaire: item.prixUnitaire,
   491→            remise: item.remise,
   492→          })),
   493→          paiements: paiementData,
   494→        }),
   495→      })
   496→
   497→      if (!res.ok) {
   498→        const err = await res.json()
   499→        throw new Error(err.error || 'Erreur')
   500→      }
   501→
   502→      const vente = await res.json()
   503→      toast.success(`Vente ${vente.reference} enregistrée`)
   504→
   505→      // Add to recent transactions
   506→      const newVente: VenteSimple = {
   507→        id: vente.id,
   508→        reference: vente.reference,
   509→        montantTotal: vente.montantTotal,
   510→        montantPaye: vente.montantPaye,
   511→        statut: vente.statut,
   512→        modePaiement: vente.modePaiement,
   513→        createdAt: vente.createdAt,
   514→        patient: vente.patient ? { id: vente.patient.id, nom: vente.patient.nom, prenom: vente.patient.prenom } : null,
   515→      }
   516→      setRecentVentes(prev => [newVente, ...prev].slice(0, 10))
   517→
   518→      // Reset POS
   519→      clearCart()
   520→    } catch (err) {
   521→      toast.error(err instanceof Error ? err.message : 'Erreur lors de la création de la vente')
   522→    } finally {
   523→      setSubmitting(false)
   524→    }
   525→  }
   526→
   527→  // ============================================================
   528→  // Create patient from POS
   529→  // ============================================================
   530→
   531→  const handleCreatePatient = async () => {
   532→    if (!newPatient.nom || !newPatient.prenom) {
   533→      toast.error('Nom et prénom sont requis')
   534→      return
   535→    }
   536→    try {
   537→      const res = await fetch('/api/patients', {
   538→        method: 'POST',
   539→        headers: { 'Content-Type': 'application/json' },
   540→        body: JSON.stringify({ pharmacieId, ...newPatient }),
   541→      })
   542→      if (!res.ok) throw new Error('Erreur')
   543→      const patient = await res.json()
   544→      setSelectedPatientId(patient.id)
   545→      setPatients(prev => [...prev, patient])
   546→      setShowNewPatient(false)
   547→      setNewPatient({ nom: '', prenom: '', telephone: '' })
   548→      toast.success('Patient créé avec succès')
   549→    } catch {
   550→      toast.error('Erreur lors de la création du patient')
   551→    }
   552→  }
   553→
   554→  // ============================================================
   555→  // Split payment helpers
   556→  // ============================================================
   557→
   558→  const addSplitPaiement = () => {
   559→    const id = String(Date.now())
   560→    setSplitPaiements(prev => [...prev, { id, mode: 'ESPECES', montant: 0, reference: '' }])
   561→  }
   562→
   563→  const removeSplitPaiement = (id: string) => {
   564→    setSplitPaiements(prev => prev.filter(p => p.id !== id).length > 0 ? prev.filter(p => p.id !== id) : prev)
   565→  }
   566→
   567→  const updateSplitPaiement = (id: string, field: 'mode' | 'montant' | 'reference', value: string | number) => {
   568→    setSplitPaiements(prev => prev.map(p =>
   569→      p.id === id ? { ...p, [field]: value } : p
   570→    ))
   571→  }
   572→
   573→  const autoFillRemaining = (id: string) => {
   574→    const otherTotal = splitPaiements.filter(p => p.id !== id).reduce((sum, p) => sum + p.montant, 0)
   575→    const remaining = Math.max(0, cartTotal - otherTotal)
