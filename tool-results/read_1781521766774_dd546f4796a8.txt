    71→  email: string | null

    82→function formatFCFA(amount: number) {
    83→  return new Intl.NumberFormat('fr-FR').format(amount) + ' FCFA'
    84→}
    85→
    86→function StarRating({ value, max = 5 }: { value: number; max?: number }) {
    87→  return (
    88→    <div className="flex gap-0.5">
    89→      {Array.from({ length: max }).map((_, i) => (
    90→        <Star
    91→          key={i}
    92→          className={`w-3.5 h-3.5 ${i < value ? 'fill-amber-400 text-amber-400' : 'text-gray-300'}`}
    93→        />
    94→      ))}
    95→    </div>
    96→  )
    97→}

   100→  const { pharmacie } = useAuth()
   101→  const [fournisseurs, setFournisseurs] = useState<Fournisseur[]>([])
   102→  const [loading, setLoading] = useState(true)
   103→  const [search, setSearch] = useState('')
   104→  const [filterType, setFilterType] = useState<string>('all')
   105→  const [filterStatus, setFilterStatus] = useState<string>('all')
   106→
   107→  // Dialogs
   108→  const [addDialogOpen, setAddDialogOpen] = useState(false)
   109→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   110→  const [detailDialogOpen, setDetailDialogOpen] = useState(false)
   111→  const [evalDialogOpen, setEvalDialogOpen] = useState(false)
   112→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)

   114→  const [detailData, setDetailData] = useState<Fournisseur | null>(null)
   115→  const [detailLoading, setDetailLoading] = useState(false)
   116→
   117→  // Add/Edit form
   118→  const [formNom, setFormNom] = useState('')

   120→  const [formAdresse, setFormAdresse] = useState('')

   122→  const [formEmail, setFormEmail] = useState('')
   123→  const [formDelai, setFormDelai] = useState('')
   124→  const [formGrossiste, setFormGrossiste] = useState(false)
   125→
   126→  // Eval form
   127→  const [evalDelai, setEvalDelai] = useState('3')
   128→  const [evalQualite, setEvalQualite] = useState('3')
   129→  const [evalCommunication, setEvalCommunication] = useState('3')
   130→  const [evalCommentaire, setEvalCommentaire] = useState('')
   131→
   132→  const fetchData = async () => {
   133→    if (!pharmacie?.id) return
   134→    setLoading(true)

   136→      const res = await fetch(`/api/fournisseurs?pharmacieId=${pharmacie.id}`)
   137→      if (res.ok) {
   138→        const data = await res.json()

   141→    } catch {
   142→      setFournisseurs([])
   143→    } finally {
   144→      setLoading(false)
   145→    }
   146→  }
   147→
   148→  useEffect(() => {
   149→    fetchData()

   158→      )

   161→      result = result.filter(f => f.estGrossisteAPI)
   162→    } else if (filterType === 'regular') {
   163→      result = result.filter(f => !f.estGrossisteAPI)
   164→    }
   165→    if (filterStatus === 'actif') {
   166→      result = result.filter(f => f.actif)

   168→      result = result.filter(f => !f.actif)
   169→    }
   170→    return result
   171→  }, [fournisseurs, search, filterType, filterStatus])
   172→

   174→    if (!pharmacie?.id || !formNom) {
   175→      toast.error('Le nom du fournisseur est requis')
   176→      return
   177→    }
   178→    try {
   179→      const res = await fetch('/api/fournisseurs', {
   180→        method: 'POST',
   181→        headers: { 'Content-Type': 'application/json' },
   182→        body: JSON.stringify({
   183→          pharmacieId: pharmacie.id,
   184→          nom: formNom,
   185→          code: formCode || `FOUR-${Date.now().toString(36).toUpperCase()}`,
   186→          adresse: formAdresse || null,
   187→          telephone: formTel || null,
   188→          email: formEmail || null,
   189→          delaiLivraison: formDelai ? parseInt(formDelai) : null,
   190→          estGrossisteAPI: formGrossiste,
   191→          actif: true,
   192→        }),
   193→      })

   195→        toast.success('Fournisseur ajouté avec succès')
   196→        setAddDialogOpen(false)
   197→        resetForm()
   198→        fetchData()
   199→      } else {
   200→        toast.error('Erreur lors de l\'ajout du fournisseur')
   201→      }
   202→    } catch {
   203→      toast.error('Erreur lors de l\'ajout du fournisseur')
   204→    }
   205→  }
   206→
   207→  const handleEditFournisseur = async () => {
   208→    if (!selectedFournisseur) return
   209→    try {
   210→      const res = await fetch(`/api/fournisseurs/${selectedFournisseur.id}`, {
   211→        method: 'PATCH',
   212→        headers: { 'Content-Type': 'application/json' },
   213→        body: JSON.stringify({
   214→          nom: formNom,
   215→          code: formCode,
   216→          adresse: formAdresse || null,
   217→          telephone: formTel || null,
   218→          email: formEmail || null,
   219→          delaiLivraison: formDelai ? parseInt(formDelai) : null,
   220→          estGrossisteAPI: formGrossiste,
   221→        }),
   222→      })
   223→      if (res.ok) {
   224→        toast.success('Fournisseur modifié avec succès')
   225→        setEditDialogOpen(false)
   226→        resetForm()
   227→        fetchData()
   228→      } else {
   229→        toast.error('Erreur lors de la modification du fournisseur')
   230→      }
   231→    } catch {
   232→      toast.error('Erreur lors de la modification du fournisseur')
   233→    }
   234→  }
   235→
   236→  const handleDeleteFournisseur = async () => {
   237→    if (!selectedFournisseur) return
   238→    try {
   239→      const res = await fetch(`/api/fournisseurs/${selectedFournisseur.id}`, {
   240→        method: 'DELETE',
   241→      })

   243→        toast.success('Fournisseur désactivé avec succès')
   244→        setDeleteDialogOpen(false)
   245→        setSelectedFournisseur(null)
   246→        fetchData()
   247→      } else {
   248→        toast.error('Erreur lors de la désactivation du fournisseur')

   250→    } catch {
   251→      toast.error('Erreur lors de la désactivation du fournisseur')
   252→    }

   256→    setSelectedFournisseur(fournisseur)

   258→    setDetailLoading(true)
   259→    try {
   260→      const res = await fetch(`/api/fournisseurs/${fournisseur.id}`)

   262→        const data = await res.json()
   263→        setDetailData(data)
   264→      }

   266→      setDetailData(fournisseur)

   268→      setDetailLoading(false)
   269→    }
   270→  }
   271→

   273→    setSelectedFournisseur(fournisseur)
   274→    setFormNom(fournisseur.nom)
   275→    setFormCode(fournisseur.code)
   276→    setFormAdresse(fournisseur.adresse || '')
   277→    setFormTel(fournisseur.telephone || '')
   278→    setFormEmail(fournisseur.email || '')

   280→    setFormGrossiste(fournisseur.estGrossisteAPI)
   281→    setEditDialogOpen(true)
   282→  }
   283→

   285→    setSelectedFournisseur(fournisseur)
   286→    setDeleteDialogOpen(true)
   287→  }
   288→
   289→  const handleOpenEval = (fournisseur: Fournisseur) => {
   290→    setSelectedFournisseur(fournisseur)
   291→    setEvalDelai('3')
   292→    setEvalQualite('3')
   293→    setEvalCommunication('3')
   294→    setEvalCommentaire('')
   295→    setEvalDialogOpen(true)
   296→  }
   297→

   299→    if (!pharmacie?.id || !selectedFournisseur) return
   300→    try {
   301→      const res = await fetch(`/api/fournisseurs/${selectedFournisseur.id}/evaluations`, {
   302→        method: 'POST',
   303→        headers: { 'Content-Type': 'application/json' },
   304→        body: JSON.stringify({
   305→          pharmacieId: pharmacie.id,
   306→          delaiRespecte: parseInt(evalDelai),
   307→          qualiteProduit: parseInt(evalQualite),
   308→          communication: parseInt(evalCommunication),
   309→          commentaire: evalCommentaire || null,
   310→        }),
   311→      })
   312→      if (res.ok) {
   313→        toast.success('Évaluation enregistrée avec succès')
   314→        setEvalDialogOpen(false)
   315→        fetchData()
   316→      } else {
   317→        toast.error('Erreur lors de l\'enregistrement de l\'évaluation')
   318→      }
   319→    } catch {
   320→      toast.error('Erreur lors de l\'enregistrement de l\'évaluation')
   321→    }
   322→  }
   323→
   324→  const resetForm = () => {
   325→    setFormNom('')
   326→    setFormCode('')
   327→    setFormAdresse('')
   328→    setFormTel('')
   329→    setFormEmail('')
   330→    setFormDelai('')
   331→    setFormGrossiste(false)
   332→  }
   333→
   334→  // KPI Calculations
   335→  const totalCount = fournisseurs.length
   336→  const activeCount = fournisseurs.filter(f => f.actif).length
   337→  const commandesEnCours = fournisseurs.reduce((sum, f) => sum + (f._count?.commandes || 0), 0)
   338→  const avgNote = fournisseurs.length > 0
   339→    ? fournisseurs.filter(f => f.noteEvaluation).length > 0
   340→      ? fournisseurs.reduce((s, f) => s + (f.noteEvaluation || 0), 0) / fournisseurs.filter(f => f.noteEvaluation).length
   341→      : 0
   342→    : 0
   343→
   344→  if (loading) {
   345→    return (
   346→      <div className="space-y-4">
   347→        <Skeleton className="h-8 w-64" />
   348→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   349→          {[1, 2, 3, 4].map(i => <Skeleton key={i} className="h-24" />)}
   350→        </div>
   351→        <Skeleton className="h-12" />
   352→        <Skeleton className="h-96" />
   353→      </div>
   354→    )
   355→  }
   356→
   357→  return (
   358→    <div className="space-y-6">
   359→      {/* Header */}
   360→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   361→        <div>
   362→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   363→            <Truck className="w-6 h-6 text-[#1D9E75]" />
   364→            Gestion des Fournisseurs
   365→          </h1>
   366→          <p className="text-sm text-muted-foreground">
   367→            {totalCount} fournisseurs référencés • {activeCount} actifs
   368→          </p>
   369→        </div>
   370→        <Button className="gap-2 bg-[#1D9E75] hover:bg-[#085041]" onClick={() => { resetForm(); setAddDialogOpen(true) }}>
   371→          <Plus className="w-4 h-4" />
   372→          Ajouter un fournisseur
   373→        </Button>
   374→      </div>
   375→
   376→      {/* KPI Cards */}
   377→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   378→        <Card>
   379→          <CardContent className="p-4">
   380→            <div className="flex items-center justify-between">
   381→              <div>
   382→                <span className="text-xs text-muted-foreground uppercase">Total Fournisseurs</span>
   383→                <span className="text-xl font-bold block">{totalCount}</span>
   384→              </div>
   385→              <Truck className="w-8 h-8 text-gray-400/30" />
   386→            </div>
   387→          </CardContent>
   388→        </Card>
   389→        <Card>
   390→          <CardContent className="p-4">
   391→            <div className="flex items-center justify-between">
   392→              <div>
   393→                <span className="text-xs text-muted-foreground uppercase">Fournisseurs Actifs</span>
   394→                <span className="text-xl font-bold block text-[#1D9E75]">{activeCount}</span>
   395→              </div>
   396→              <Users className="w-8 h-8 text-[#1D9E75]/30" />
   397→            </div>
   398→          </CardContent>
   399→        </Card>
   400→        <Card>
   401→          <CardContent className="p-4">
   402→            <div className="flex items-center justify-between">
   403→              <div>
   404→                <span className="text-xs text-muted-foreground uppercase">Note Moyenne</span>
   405→                <span className="text-xl font-bold block">{avgNote > 0 ? avgNote.toFixed(1) + ' / 5' : 'N/A'}</span>
   406→              </div>
   407→              <Star className="w-8 h-8 text-amber-400/30" />
   408→            </div>
   409→          </CardContent>
   410→        </Card>
   411→        <Card>
   412→          <CardContent className="p-4">
   413→            <div className="flex items-center justify-between">
   414→              <div>
   415→                <span className="text-xs text-muted-foreground uppercase">Commandes en cours</span>
   416→                <span className="text-xl font-bold block text-amber-600">{commandesEnCours}</span>
   417→              </div>
   418→              <Package className="w-8 h-8 text-amber-400/30" />
   419→            </div>
   420→          </CardContent>
   421→        </Card>
   422→      </div>
   423→
   424→      {/* Filters */}
   425→      <Card>
   426→        <CardContent className="p-4">
   427→          <div className="flex flex-col sm:flex-row gap-3">
   428→            <div className="relative flex-1">
   429→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   430→              <Input
   431→                placeholder="Rechercher par nom, code..."
   432→                className="pl-9"
   433→                value={search}
   434→                onChange={e => setSearch(e.target.value)}
   435→              />
   436→            </div>
   437→            <Select value={filterStatus} onValueChange={setFilterStatus}>
   438→              <SelectTrigger className="w-44">
   439→                <SelectValue />
   440→              </SelectTrigger>
   441→              <SelectContent>
   442→                <SelectItem value="all">Tous les statuts</SelectItem>
   443→                <SelectItem value="actif">Actifs</SelectItem>
   444→                <SelectItem value="inactif">Inactifs</SelectItem>
   445→              </SelectContent>
   446→            </Select>
   447→            <Select value={filterType} onValueChange={setFilterType}>
   448→              <SelectTrigger className="w-48">
   449→                <Filter className="w-4 h-4 mr-2" />
   450→                <SelectValue />
   451→              </SelectTrigger>
   452→              <SelectContent>
   453→                <SelectItem value="all">Tous les types</SelectItem>
   454→                <SelectItem value="grossiste">Grossistes API</SelectItem>
   455→                <SelectItem value="regular">Fournisseurs classiques</SelectItem>
   456→              </SelectContent>
   457→            </Select>
   458→          </div>
   459→        </CardContent>
   460→      </Card>
   461→
   462→      {/* Table */}
   463→      <Card>
   464→        <CardContent className="p-0">
   465→          <div className="overflow-x-auto">
   466→            <Table>
   467→              <TableHeader>
   468→                <TableRow>
   469→                  <TableHead>Code</TableHead>
   470→                  <TableHead>Nom</TableHead>
   471→                  <TableHead>Téléphone</TableHead>
   472→                  <TableHead>Email</TableHead>
   473→                  <TableHead className="text-center">Délai Livraison</TableHead>
   474→                  <TableHead className="text-center">Note</TableHead>
   475→                  <TableHead className="text-center">Statut</TableHead>
   476→                  <TableHead className="text-right">Actions</TableHead>
   477→                </TableRow>
   478→              </TableHeader>
   479→              <TableBody>
   480→                {filteredFournisseurs.length === 0 ? (
   481→                  <TableRow>
   482→                    <TableCell colSpan={8} className="text-center py-8 text-muted-foreground">
   483→                      Aucun fournisseur trouvé
   484→                    </TableCell>
   485→                  </TableRow>
   486→                ) : (
   487→                  filteredFournisseurs.map(f => (
   488→                    <TableRow key={f.id} className="hover:bg-muted/30">
   489→                      <TableCell>
   490→                        <span className="text-xs font-mono bg-muted px-1.5 py-0.5 rounded">{f.code}</span>
   491→                      </TableCell>
   492→                      <TableCell>
   493→                        <div className="flex flex-col">
   494→                          <span className="font-medium text-sm">{f.nom}</span>
   495→                          {f.estGrossisteAPI && (

   498→                        </div>
   499→                      </TableCell>
   500→                      <TableCell className="text-sm">{f.telephone || '—'}</TableCell>
   501→                      <TableCell className="text-sm text-muted-foreground">{f.email || '—'}</TableCell>

   503→                        {f.delaiLivraison ? `${f.delaiLivraison} jours` : '—'}
   504→                      </TableCell>

   506→                        {f.noteEvaluation ? (

   513→                        )}
   514→                      </TableCell>
   515→                      <TableCell className="text-center">
   516→                        {f.actif ? (
   517→                          <Badge className="text-[9px] bg-[#E1F5EE] text-[#085041] border-0">Actif</Badge>

   529→                          </Button>
   530→                          <Button size="sm" variant="ghost" className="h-8 w-8 p-0" onClick={() => handleOpenEval(f)} title="Évaluer">
   531→                            <Star className="w-4 h-4" />
   532→                          </Button>
   533→                          <Button size="sm" variant="ghost" className="h-8 w-8 p-0 text-red-500 hover:text-red-700" onClick={() => handleOpenDelete(f)} title="Désactiver">
   534→                            <Trash2 className="w-4 h-4" />
   535→                          </Button>
   536→                        </div>
   537→                      </TableCell>
   538→                    </TableRow>
   539→                  ))
   540→                )}
   541→              </TableBody>
   542→            </Table>
   543→          </div>
   544→        </CardContent>
   545→      </Card>
   546→
   547→      {/* Add Supplier Dialog */}
   548→      <Dialog open={addDialogOpen} onOpenChange={setAddDialogOpen}>
