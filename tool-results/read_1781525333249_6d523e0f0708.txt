  9007→// Vérification signature RSA-256 des alertes DPMED
  9008→async verifyDPMEDSignature(payload: any, signature: string): Promise<boolean> {
  9009→const publicKey = await this.keyStore.getDPMEDPublicKey();
  9010→const verify = crypto.createVerify('SHA256');
  9011→verify.update(JSON.stringify(payload));
  9012→return verify.verify(publicKey, signature, 'base64');
  9013→}
