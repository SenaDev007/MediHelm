   472→  const interactions: Interaction[] = [];

   483→        x.dci.toLowerCase().includes(dcis[j].toLowerCase())
   484→      );

   486→      if (inter && (inter.gravite === 'MAJEURE' || inter.gravite === 'MODEREE')) {

   488→          dci1: dcis[i], dci2: dcis[j],
   489→          gravite: inter.gravite,
   490→          description: inter.description
   491→        });
   492→      }
   493→    }
   494→  }
   495→  return interactions; // vide = pas d'interaction connue

   605→async diffuserAlerteComplete(alerteId: string) {
   606→  const alerte = await this.prisma.alerteDPMED.findUniqueOrThrow({
   607→    where: { id: alerteId }
   608→  });
   609→
   610→  // 1. Vérifier la signature numérique de l'alerte
   611→  const valid = await this.verifierSignature(alerte.signatureNumerique, alerte);
   612→  if (!valid) throw new Error('MH-SEC-001: Signature alerte invalide — diffusion annulée');
   613→
   614→  // 2. Identifier les pharmacies concernées
   615→  //    (ayant le lot concerné en stock actif)
   616→  const pharmaciesConcernees = await this.prisma.$queryRaw`
   617→    SELECT DISTINCT l.pharmacie_id, array_agg(l.numero_lot) as lots
   618→    FROM lots l
   619→    JOIN medicaments m ON l.medicament_id = m.id
   620→    WHERE l.quantite > 0
   621→    AND m.dci ILIKE ${alerte.dciConcernee}
   622→    AND l.numero_lot = ANY(${alerte.numerosLotConcernes})
   623→    GROUP BY l.pharmacie_id
   624→  `;
   625→
   626→  // 3. Identifier les patients concernés
   627→  //    (ayant acheté le lot dans les 90 derniers jours)
   628→  const patientsConcernes = await this.getPatientsConcernes(alerte, 90);
   629→
   630→  // 4. Diffusion multi-canal aux pharmaciens
   631→  for (const pharmacie of pharmaciesConcernees) {
   632→    await this.prisma.diffusionAlerte.create({
   633→      data: { alerteId, pharmacieId: pharmacie.pharmacie_id,
   634→              lotsConcernes: pharmacie.lots,
   635→              canalEnvoi: ['PUSH', 'SMS'] }
   636→    });
   637→    await this.notifService.pushAlerteDPMED(pharmacie.pharmacie_id, alerte);
   638→    await this.smsService.sendAlerteDPMED(pharmacie.pharmacie_id, alerte);
   639→  }
   640→
   641→  // 5. Diffusion aux patients concernés
   642→  for (const patient of patientsConcernes) {
   643→    await this.notifService.pushAlertePatient(patient, alerte);
   644→    if (patient.telephone) await this.smsService.sendAlertePatient(patient, alerte);
   645→  }
   646→
   647→  // 6. Mise à jour compteurs
   648→  await this.prisma.alerteDPMED.update({
   649→    where: { id: alerteId },
   650→    data: {
   651→      statut: 'DIFFUSEE',
   652→      nbOfficinesNotifiees: pharmaciesConcernees.length,
   653→      nbPatientsNotifies: patientsConcernes.length
   654→    }
   655→  });
   656→}

   680→// En l'absence d'API DPMED : saisie manuelle via Admin MédiHelm
   681→// + confirmation humaine obligatoire avant mise en queue
   682→async calculerScoreConformite(pharmacieId: string): Promise<ScoreConformite> {
   683→
   684→  // 1. Registre stupéfiants (25 pts)
   685→  //    Vérifie que toutes les délivrances de stupéfiants ont un numéro de registre
   686→  const stupMissing = await this.prisma.vente.count({
   687→    where: { pharmacieId,
   688→             lignes: { some: { medicament: { estStupefiant: true } } },
   689→             ordonnanceId: null  // sans ordonnance liée
   690→           }
   691→  });
   692→  const scoreStup = stupMissing === 0 ? 25 : Math.max(0, 25 - stupMissing * 5);
   693→
   694→  // 2. Alertes DPMED traitées < 24h (25 pts)
   695→  const alertes = await this.prisma.diffusionAlerte.findMany({
   696→    where: { pharmacieId, dateEnvoi: { gte: subDays(new Date(), 90) } }
   697→  });
   698→  const traitees24h = alertes.filter(a =>
   699→    a.dateAcquittement &&
   700→    differenceInHours(a.dateAcquittement, a.dateEnvoi) <= 24
   701→  ).length;
   702→  const scoreAlertes = alertes.length === 0 ? 25
   703→    : Math.round((traitees24h / alertes.length) * 25);
   704→
   705→  // 3. Documents valides (20 pts)
   706→  const docsExpires = await this.prisma.documentReglementaire.count({
   707→    where: { pharmacieId, statut: { in: ['EXPIRE', 'EXPIRE_BIENTOT'] } }
   708→  });
   709→  const scoreDocs = docsExpires === 0 ? 20 : Math.max(0, 20 - docsExpires * 5);
   710→
   711→  // 4. Signalements EI soumis (15 pts)
   712→  const eiNonSoumis = await this.prisma.signalementEI.count({
   713→    where: { pharmacieId, statutEnvoi: 'EN_ATTENTE',
   714→             createdAt: { lte: subDays(new Date(), 7) } }  // > 7j non soumis
   715→  });
   716→  const scoreEI = eiNonSoumis === 0 ? 15 : Math.max(0, 15 - eiNonSoumis * 5);
   717→
   718→  // 5. PV destructions à jour (15 pts)
   719→  const destructSansVP = await this.prisma.destructionMedicament.count({
   720→    where: { pharmacieId, pvUrl: null }
   721→  });
   722→  const scoreDestr = destructSansVP === 0 ? 15 : Math.max(0, 15 - destructSansVP * 5);
   723→

   782→async verifyDPMEDSignature(payload: any, signature: string): Promise<boolean> {
   783→  const publicKey = await this.keyStore.getDPMEDPublicKey();
   784→  const verify = crypto.createVerify('SHA256');
   785→  verify.update(JSON.stringify(payload));
   786→  return verify.verify(publicKey, signature, 'base64');
   787→}
