   132→  new Intl.NumberFormat('fr-FR').format(Math.round(amount)) + ' FCFA'
   133→
   134→const formatDate = (dateStr: string) => {
   135→  const d = new Date(dateStr)
   136→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric' })
   137→}
   138→
   139→const formatDateTime = (dateStr: string) => {
   140→  const d = new Date(dateStr)
   141→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric' }) +
   142→    ' ' + d.toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })
   143→}

   203→export default function VentesPage() {
   204→  const { pharmacie } = useAuth()
   205→  const pharmacieId = pharmacie?.id || ''
   206→
   207→  // Data state
   208→  const [ventes, setVentes] = useState<Vente[]>([])
   209→  const [total, setTotal] = useState(0)
   210→  const [stats, setStats] = useState({ caDuJour: 0, nbVentesJour: 0, panierMoyen: 0, ventesEnAttente: 0 })
   211→  const [loading, setLoading] = useState(true)
   212→
   213→  // Filter state
   214→  const [page, setPage] = useState(1)
   215→  const [search, setSearch] = useState('')
   216→  const [filterStatut, setFilterStatut] = useState<string>('')
   217→  const [filterMode, setFilterMode] = useState<string>('')
   218→  const [filterDateDebut, setFilterDateDebut] = useState('')
   219→  const [filterDateFin, setFilterDateFin] = useState('')
   220→  const [sortBy, setSortBy] = useState<'createdAt' | 'montantTotal'>('createdAt')
   221→  const [sortOrder, setSortOrder] = useState<'asc' | 'desc'>('desc')
   222→  const [showFilters, setShowFilters] = useState(false)
   223→
   224→  // Dialog state
   225→  const [selectedVente, setSelectedVente] = useState<Vente | null>(null)
   226→  const [detailOpen, setDetailOpen] = useState(false)
   227→  const [posOpen, setPosOpen] = useState(false)
   228→
   229→  // POS state
   230→  const [cart, setCart] = useState<CartItem[]>([])
   231→  const [medSearch, setMedSearch] = useState('')
   232→  const [medicaments, setMedicaments] = useState<Medicament[]>([])
   233→  const [selectedPatientId, setSelectedPatientId] = useState<string>('')
   234→  const [patients, setPatients] = useState<Patient[]>([])
   235→  const [patientSearch, setPatientSearch] = useState('')
   236→  const [posModePaiement, setPosModePaiement] = useState<ModePaiement>('ESPECES')
   237→  const [posGlobalRemise, setPosGlobalRemise] = useState(0)
   238→  const [submitting, setSubmitting] = useState(false)
   239→  const [showNewPatient, setShowNewPatient] = useState(false)
   240→  const [newPatient, setNewPatient] = useState({ nom: '', prenom: '', telephone: '' })
   241→
   242→  const limit = 20
   243→
   244→  // ============================================================
   245→  // Fetch ventes
   246→  // ============================================================
   247→
   248→  const fetchVentes = useCallback(async () => {
   249→    if (!pharmacieId) return
   250→    setLoading(true)
   251→    try {
   252→      const params = new URLSearchParams({
   253→        pharmacieId,
   254→        page: String(page),
   255→        limit: String(limit),
   256→        sortBy,
   257→        sortOrder,
   258→      })
   259→      if (search) params.set('search', search)
   260→      if (filterStatut) params.set('statut', filterStatut)
   261→      if (filterMode) params.set('modePaiement', filterMode)
   262→      if (filterDateDebut) params.set('dateDebut', filterDateDebut)
   263→      if (filterDateFin) params.set('dateFin', filterDateFin)
   264→
   265→      const res = await fetch(`/api/ventes?${params}`)
   266→      if (!res.ok) throw new Error('Erreur réseau')
   267→      const data = await res.json()
   268→      setVentes(data.ventes || [])
   269→      setTotal(data.total || 0)
   270→      if (data.stats) setStats(data.stats)
   271→    } catch {
   272→      toast.error('Erreur lors du chargement des ventes')
   273→    } finally {
   274→      setLoading(false)
   275→    }
   276→  }, [pharmacieId, page, search, filterStatut, filterMode, filterDateDebut, filterDateFin, sortBy, sortOrder])
   277→
   278→  useEffect(() => {
   279→    fetchVentes()
   280→  }, [fetchVentes])
   281→
   282→  // ============================================================
   283→  // Fetch medicaments for POS
   284→  // ============================================================
   285→
   286→  const fetchMedicaments = useCallback(async () => {
   287→    if (!pharmacieId || !medSearch) return
   288→    try {
   289→      const res = await fetch(`/api/medicaments?pharmacieId=${pharmacieId}&search=${encodeURIComponent(medSearch)}`)
   290→      if (res.ok) {
   291→        const data = await res.json()
   292→        setMedicaments(data)
   293→      }
   294→    } catch {
   295→      // silent
   296→    }
   297→  }, [pharmacieId, medSearch])
   298→
   299→  useEffect(() => {
   300→    const timeout = setTimeout(fetchMedicaments, 300)
   301→    return () => clearTimeout(timeout)
   302→  }, [fetchMedicaments])
   303→
   304→  // ============================================================
   305→  // Fetch patients for POS
   306→  // ============================================================
   307→
   308→  const fetchPatients = useCallback(async () => {
   309→    if (!pharmacieId) return
   310→    try {
   311→      const params = new URLSearchParams({ pharmacieId })
   312→      if (patientSearch) params.set('search', patientSearch)
   313→      const res = await fetch(`/api/patients?${params}`)
   314→      if (res.ok) {
   315→        const data = await res.json()
   316→        setPatients(data)
   317→      }
   318→    } catch {
   319→      // silent
   320→    }
   321→  }, [pharmacieId, patientSearch])
   322→
   323→  useEffect(() => {
   324→    const timeout = setTimeout(fetchPatients, 300)
   325→    return () => clearTimeout(timeout)
   326→  }, [fetchPatients])
   327→
   328→  // ============================================================
   329→  // Cart calculations
   330→  // ============================================================
   331→
   332→  const cartTotal = useMemo(() => {
   333→    const subtotal = cart.reduce((sum, item) => sum + (item.prixUnitaire * item.quantite - item.remise), 0)
   334→    return Math.max(0, subtotal - posGlobalRemise)
   335→  }, [cart, posGlobalRemise])
   336→
   337→  // ============================================================
   338→  // Cart actions
   339→  // ============================================================
   340→
   341→  const addToCart = (med: Medicament) => {
   342→    setCart(prev => {
   343→      const existing = prev.find(i => i.medicamentId === med.id)
   344→      if (existing) {
   345→        return prev.map(i =>
   346→          i.medicamentId === med.id
   347→            ? { ...i, quantite: i.quantite + 1 }
   348→            : i
   349→        )
   350→      }
   351→      return [...prev, {
   352→        medicamentId: med.id,
   353→        nomCommercial: med.nomCommercial,
   354→        dci: med.dci,
   355→        prixUnitaire: med.prixPublic,
   356→        quantite: 1,
   357→        remise: 0,
   358→      }]
   359→    })
   360→    toast.success(`${med.nomCommercial} ajouté au panier`)
   361→  }
   362→
   363→  const removeFromCart = (medicamentId: string) => {
   364→    setCart(prev => prev.filter(i => i.medicamentId !== medicamentId))
   365→  }
   366→
   367→  const updateCartItem = (medicamentId: string, field: 'quantite' | 'remise', value: number) => {
   368→    setCart(prev => prev.map(i =>
   369→      i.medicamentId === medicamentId
   370→        ? { ...i, [field]: Math.max(0, value) }
   371→        : i
   372→    ))
   373→  }
   374→
   375→  // ============================================================
   376→  // Submit sale
   377→  // ============================================================
   378→
   379→  const handleSubmitSale = async () => {
   380→    if (cart.length === 0) {
   381→      toast.error('Ajoutez au moins un médicament')
   382→      return
   383→    }
   384→    setSubmitting(true)
   385→    try {
   386→      const res = await fetch('/api/ventes', {
   387→        method: 'POST',
   388→        headers: { 'Content-Type': 'application/json' },
   389→        body: JSON.stringify({
   390→          pharmacieId,
   391→          patientId: selectedPatientId || null,
   392→          modePaiement: posModePaiement,
   393→          remise: posGlobalRemise,
   394→          lignes: cart.map(item => ({
   395→            medicamentId: item.medicamentId,
   396→            quantite: item.quantite,
   397→            prixUnitaire: item.prixUnitaire,
   398→            remise: item.remise,
   399→          })),
   400→        }),
   401→      })
   402→      if (!res.ok) {
   403→        const err = await res.json()
   404→        throw new Error(err.error || 'Erreur lors de la création')
   405→      }
   406→      toast.success('Vente enregistrée avec succès')
   407→      setPosOpen(false)
   408→      resetPos()
   409→      fetchVentes()
   410→    } catch (err) {
   411→      toast.error(err instanceof Error ? err.message : 'Erreur lors de la création de la vente')
   412→    } finally {
   413→      setSubmitting(false)
   414→    }
   415→  }
   416→
   417→  // ============================================================
   418→  // Create patient from POS
   419→  // ============================================================
   420→
   421→  const handleCreatePatient = async () => {
   422→    if (!newPatient.nom || !newPatient.prenom) {
   423→      toast.error('Nom et prénom sont requis')
   424→      return
   425→    }
   426→    try {
   427→      const res = await fetch('/api/patients', {
   428→        method: 'POST',
   429→        headers: { 'Content-Type': 'application/json' },
   430→        body: JSON.stringify({
   431→          pharmacieId,
   432→          ...newPatient,
   433→        }),
   434→      })
   435→      if (!res.ok) throw new Error('Erreur')
   436→      const patient = await res.json()
   437→      setSelectedPatientId(patient.id)
   438→      setPatients(prev => [...prev, patient])
   439→      setShowNewPatient(false)
   440→      setNewPatient({ nom: '', prenom: '', telephone: '' })
   441→      toast.success('Patient créé avec succès')
   442→    } catch {
   443→      toast.error('Erreur lors de la création du patient')
   444→    }
   445→  }
   446→
   447→  // ============================================================
   448→  // Cancel sale
   449→  // ============================================================
   450→
   451→  const handleCancelVente = async (venteId: string) => {
   452→    try {
   453→      const res = await fetch(`/api/ventes/${venteId}`, {
   454→        method: 'PATCH',
   455→        headers: { 'Content-Type': 'application/json' },
   456→        body: JSON.stringify({ statut: 'ANNULEE' }),
   457→      })
   458→      if (!res.ok) throw new Error('Erreur')
   459→      toast.success('Vente annulée')
   460→      setDetailOpen(false)
   461→      fetchVentes()
   462→    } catch {
   463→      toast.error('Erreur lors de l\'annulation')
   464→    }
   465→  }
   466→
   467→  // ============================================================
   468→  // Reset POS
   469→  // ============================================================
   470→
   471→  const resetPos = () => {
   472→    setCart([])
   473→    setMedSearch('')
   474→    setMedicaments([])
   475→    setSelectedPatientId('')
   476→    setPatientSearch('')
   477→    setPosModePaiement('ESPECES')
   478→    setPosGlobalRemise(0)
   479→    setShowNewPatient(false)
   480→    setNewPatient({ nom: '', prenom: '', telephone: '' })
   481→  }
   482→
   483→  // ============================================================
   484→  // Open detail
   485→  // ============================================================
   486→
   487→  const openDetail = async (vente: Vente) => {
   488→    try {
   489→      const res = await fetch(`/api/ventes/${vente.id}`)
   490→      if (res.ok) {
   491→        const full = await res.json()
   492→        setSelectedVente(full)
   493→      } else {
   494→        setSelectedVente(vente)
   495→      }
   496→    } catch {
   497→      setSelectedVente(vente)
   498→    }
   499→    setDetailOpen(true)
   500→  }
   501→
   502→  // ============================================================
   503→  // Sort toggle
   504→  // ============================================================
   505→
   506→  const toggleSort = (field: 'createdAt' | 'montantTotal') => {
   507→    if (sortBy === field) {
   508→      setSortOrder(prev => prev === 'asc' ? 'desc' : 'asc')
   509→    } else {
   510→      setSortBy(field)
   511→      setSortOrder('desc')
   512→    }
   513→    setPage(1)
   514→  }
   515→
   516→  const totalPages = Math.ceil(total / limit)
   517→
   518→  // ============================================================
   519→  // Render
   520→  // ============================================================
   521→
   522→  return (
   523→    <TooltipProvider>
   524→      <div className="space-y-6">
   525→        {/* Header */}
   526→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   527→          <div>
   528→            <h1 className="text-2xl font-bold tracking-tight">Ventes</h1>
   529→            <p className="text-muted-foreground text-sm">Gestion des ventes et caisse</p>
   530→          </div>
   531→          <Button
   532→            onClick={() => { resetPos(); setPosOpen(true) }}
   533→            className="bg-[#1D9E75] hover:bg-[#178a65] text-white gap-2"
   534→          >
   535→            <Plus className="h-4 w-4" />
   536→            Nouvelle vente
   537→          </Button>
   538→        </div>
   539→
   540→        {/* Overview Cards */}
   541→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   542→          <Card className="border-l-4 border-l-[#1D9E75]">
   543→            <CardContent className="p-4">
   544→              <div className="flex items-center justify-between">
   545→                <div>
   546→                  <p className="text-sm text-muted-foreground">CA du jour</p>
   547→                  <p className="text-xl font-bold mt-1">{formatFCFA(stats.caDuJour)}</p>
   548→                </div>
   549→                <div className="h-10 w-10 rounded-lg bg-emerald-50 flex items-center justify-center">
   550→                  <DollarSign className="h-5 w-5 text-[#1D9E75]" />
   551→                </div>
   552→              </div>
   553→            </CardContent>
   554→          </Card>
   555→          <Card className="border-l-4 border-l-emerald-500">
   556→            <CardContent className="p-4">
   557→              <div className="flex items-center justify-between">
   558→                <div>
   559→                  <p className="text-sm text-muted-foreground">Ventes du jour</p>
   560→                  <p className="text-xl font-bold mt-1">{stats.nbVentesJour}</p>
   561→                </div>
   562→                <div className="h-10 w-10 rounded-lg bg-emerald-50 flex items-center justify-center">
   563→                  <ShoppingCart className="h-5 w-5 text-emerald-600" />
   564→                </div>
   565→              </div>
   566→            </CardContent>
   567→          </Card>
   568→          <Card className="border-l-4 border-l-amber-500">
   569→            <CardContent className="p-4">
   570→              <div className="flex items-center justify-between">
   571→                <div>
   572→                  <p className="text-sm text-muted-foreground">Panier moyen</p>
   573→                  <p className="text-xl font-bold mt-1">{formatFCFA(stats.panierMoyen)}</p>
   574→                </div>
   575→                <div className="h-10 w-10 rounded-lg bg-amber-50 flex items-center justify-center">

   579→            </CardContent>
