   115→// === Helpers ===

   117→function formatDate(dateStr: string | null): string {
   118→  if (!dateStr) return '—'
   119→  try {
   120→    return new Date(dateStr).toLocaleDateString('fr-FR', {
   121→      day: '2-digit',
   122→      month: '2-digit',
   123→      year: 'numeric',
   124→    })
   125→  } catch {
   126→    return dateStr
   127→  }
   128→}
   129→
   130→function formatFCFA(amount: number): string {
   131→  return new Intl.NumberFormat('fr-FR', {
   132→    style: 'decimal',
   133→    minimumFractionDigits: 0,
   134→    maximumFractionDigits: 0,
   135→  }).format(amount) + ' FCFA'
   136→}

   138→function statutCommandeLabel(statut: StatutCommande): string {
   139→  const labels: Record<StatutCommande, string> = {
   140→    BROUILLON: 'Brouillon',
   141→    ENVOYEE: 'Envoyée',
   142→    CONFIRMEE: 'Confirmée',
   143→    EN_PREPARATION: 'En préparation',
   144→    LIVREE_PARTIELLEMENT: 'Partiellement reçue',
   145→    LIVREE: 'Reçue',
   146→    ANNULEE: 'Annulée',
   147→  }
   148→  return labels[statut] || statut
   149→}
   150→
   151→function statutCommandeBadge(statut: StatutCommande) {
   152→  const colorMap: Record<StatutCommande, string> = {
   153→    BROUILLON: 'bg-gray-100 text-gray-700 border-gray-200',
   154→    ENVOYEE: 'bg-blue-50 text-blue-700 border-blue-200',
   155→    CONFIRMEE: 'bg-cyan-50 text-cyan-700 border-cyan-200',
   156→    EN_PREPARATION: 'bg-amber-50 text-amber-700 border-amber-200',
   157→    LIVREE_PARTIELLEMENT: 'bg-orange-50 text-orange-700 border-orange-200',
   158→    LIVREE: 'bg-green-50 text-green-700 border-green-200',
   159→    ANNULEE: 'bg-red-50 text-red-700 border-red-200',
   160→  }
   161→  return (
   162→    <Badge variant="outline" className={`${colorMap[statut]} border`}>
   163→      {statutCommandeLabel(statut)}
   164→    </Badge>
   165→  )
   166→}
