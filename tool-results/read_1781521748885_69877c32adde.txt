    97→export default function RetoursPage() {
    98→  const { pharmacie, user } = useAuth()
    99→  const [activeTab, setActiveTab] = useState('retours')
   100→
   101→  const [retours, setRetours] = useState<Retour[]>([])
   102→  const [destructions, setDestructions] = useState<Destruction[]>([])
   103→  const [medicaments, setMedicaments] = useState<Medicament[]>([])
   104→  const [fournisseurs, setFournisseurs] = useState<Fournisseur[]>([])
   105→  const [utilisateurs, setUtilisateurs] = useState<Utilisateur[]>([])
   106→  const [loading, setLoading] = useState(true)
   107→  const [search, setSearch] = useState('')
   108→
   109→  // Filters
   110→  const [filterRetourType, setFilterRetourType] = useState<string>('all')
   111→  const [filterRetourStatut, setFilterRetourStatut] = useState<string>('all')
   112→  const [filterDestructionStatut, setFilterDestructionStatut] = useState<string>('all')
   113→
   114→  // Dialogs
   115→  const [addRetourDialogOpen, setAddRetourDialogOpen] = useState(false)
   116→  const [addDestructionDialogOpen, setAddDestructionDialogOpen] = useState(false)
   117→  const [traiterRetourDialogOpen, setTraiterRetourDialogOpen] = useState(false)
   118→  const [updateDestructionDialogOpen, setUpdateDestructionDialogOpen] = useState(false)
   119→  const [selectedRetour, setSelectedRetour] = useState<Retour | null>(null)
   120→  const [selectedDestruction, setSelectedDestruction] = useState<Destruction | null>(null)
   121→
   122→  // Retour form
   123→  const [retMedicamentId, setRetMedicamentId] = useState('')
   124→  const [retLotId, setRetLotId] = useState('')
   125→  const [retType, setRetType] = useState('PERIME')
   126→  const [retQuantite, setRetQuantite] = useState('')
   127→  const [retMotif, setRetMotif] = useState('')
   128→  const [retFournisseurId, setRetFournisseurId] = useState('')
   129→
   130→  // Destruction form
   131→  const [destMedicamentId, setDestMedicamentId] = useState('')
   132→  const [destLotId, setDestLotId] = useState('')
   133→  const [destQuantite, setDestQuantite] = useState('')
   134→  const [destMotif, setDestMotif] = useState('')
   135→  const [destDate, setDestDate] = useState(new Date().toISOString().slice(0, 10))
   136→  const [destPharmacienId, setDestPharmacienId] = useState('')
   137→  const [destTemoinId, setDestTemoinId] = useState('')
   138→
   139→  const fetchAllData = async () => {
   140→    if (!pharmacie?.id) return
   141→    setLoading(true)
   142→    try {
   143→      const [retRes, destRes, medRes, fourRes, utilRes] = await Promise.all([
   144→        fetch(`/api/retours?pharmacieId=${pharmacie.id}`),
   145→        fetch(`/api/destructions?pharmacieId=${pharmacie.id}`),
   146→        fetch(`/api/medicaments?pharmacieId=${pharmacie.id}`),
   147→        fetch(`/api/fournisseurs?pharmacieId=${pharmacie.id}`),
   148→        fetch(`/api/utilisateurs?pharmacieId=${pharmacie.id}`),
   149→      ])
   150→      if (retRes.ok) setRetours(await retRes.json())
   151→      if (destRes.ok) setDestructions(await destRes.json())
   152→      if (medRes.ok) setMedicaments(await medRes.json())
   153→      if (fourRes.ok) setFournisseurs(await fourRes.json())
   154→      if (utilRes.ok) setUtilisateurs(await utilRes.json())
   155→
   156→      // Set current user as pharmacien
   157→      if (user) {
   158→        setDestPharmacienId(user.id)
   159→      }
   160→    } catch {
   161→      // silent
   162→    } finally {
   163→      setLoading(false)
   164→    }
   165→  }
   166→
   167→  useEffect(() => {
   168→    fetchAllData()
   169→  }, [pharmacie?.id])
   170→
   171→  // Selected medicament lots
   172→  const selectedRetMed = useMemo(() => medicaments.find(m => m.id === retMedicamentId), [medicaments, retMedicamentId])
   173→  const selectedDestMed = useMemo(() => medicaments.find(m => m.id === destMedicamentId), [medicaments, destMedicamentId])
   174→
   175→  const filteredRetours = useMemo(() => {
   176→    let result = [...retours]
   177→    if (search) {
   178→      const q = search.toLowerCase()
   179→      result = result.filter(r =>
   180→        r.medicament.nomCommercial.toLowerCase().includes(q) ||
   181→        r.motif.toLowerCase().includes(q) ||
   182→        (r.lot?.numeroLot || '').toLowerCase().includes(q)
   183→      )
   184→    }
   185→    if (filterRetourType !== 'all') {
   186→      result = result.filter(r => r.typeRetour === filterRetourType)
   187→    }
   188→    if (filterRetourStatut !== 'all') {
   189→      result = result.filter(r => r.statut === filterRetourStatut)
   190→    }
   191→    return result
   192→  }, [retours, search, filterRetourType, filterRetourStatut])
   193→
   194→  const filteredDestructions = useMemo(() => {
   195→    let result = [...destructions]
   196→    if (search) {
   197→      const q = search.toLowerCase()
   198→      result = result.filter(d =>
   199→        d.medicament.nomCommercial.toLowerCase().includes(q) ||
   200→        d.motif.toLowerCase().includes(q) ||
   201→        d.lot.numeroLot.toLowerCase().includes(q)
   202→      )
   203→    }
   204→    if (filterDestructionStatut !== 'all') {
   205→      result = result.filter(d => d.statut === filterDestructionStatut)
   206→    }
   207→    return result
   208→  }, [destructions, search, filterDestructionStatut])
   209→
   210→  // KPI calculations
   211→  const now = new Date()
   212→  const thisMonth = retours.filter(r => {
   213→    const d = new Date(r.createdAt)
   214→    return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear()
   215→  })
   216→  const retoursCeMois = thisMonth.length
   217→  const valeurRetours = thisMonth.reduce((sum, r) => {
   218→    const prix = (r.medicament as { prixVente?: number }).prixVente || 0
   219→    return sum + (prix * r.quantite)
   220→  }, 0)
   221→  const enAttente = retours.filter(r => r.statut === 'EN_COURS').length
   222→  const traites = retours.filter(r => r.statut === 'TRAITE').length
   223→
   224→  const getTypeRetourBadge = (type: string) => {
   225→    const map: Record<string, { label: string; className: string }> = {
   226→      PERIME: { label: 'Périmé', className: 'bg-red-100 text-red-800 border-0' },
   227→      ENDOMMAGE: { label: 'Endommagé', className: 'bg-orange-100 text-orange-800 border-0' },
   228→      ERREUR_LIVRAISON: { label: 'Erreur livraison', className: 'bg-blue-100 text-blue-800 border-0' },
   229→      RAPPEL_LOT: { label: 'Rappel de lot', className: 'bg-purple-100 text-purple-800 border-0' },
   230→    }
   231→    const info = map[type] || { label: type, className: 'bg-gray-100 text-gray-800 border-0' }
   232→    return <Badge className={`text-[9px] ${info.className}`}>{info.label}</Badge>
   233→  }
   234→
   235→  const getStatutRetourBadge = (statut: string) => {
   236→    const map: Record<string, { label: string; className: string }> = {
   237→      EN_COURS: { label: 'En attente', className: 'bg-amber-100 text-amber-800 border-0' },
   238→      TRAITE: { label: 'Traité', className: 'bg-[#E1F5EE] text-[#085041] border-0' },
   239→      ANNULE: { label: 'Annulé', className: 'bg-gray-100 text-gray-600 border-0' },
   240→    }
   241→    const info = map[statut] || { label: statut, className: 'bg-gray-100 text-gray-800 border-0' }
   242→    return <Badge className={`text-[9px] ${info.className}`}>{info.label}</Badge>
   243→  }
   244→
   245→  const getStatutDestructionBadge = (statut: string) => {
   246→    const map: Record<string, { label: string; className: string }> = {
   247→      PLANIFIEE: { label: 'Planifiée', className: 'bg-blue-100 text-blue-800 border-0' },
   248→      EN_COURS: { label: 'En cours', className: 'bg-amber-100 text-amber-800 border-0' },
   249→      EFFECTUEE: { label: 'Effectuée', className: 'bg-[#E1F5EE] text-[#085041] border-0' },
   250→      PV_SIGNE: { label: 'PV signé', className: 'bg-[#1D9E75] text-white border-0' },
   251→    }
   252→    const info = map[statut] || { label: statut, className: 'bg-gray-100 text-gray-800 border-0' }
   253→    return <Badge className={`text-[9px] ${info.className}`}>{info.label}</Badge>
   254→  }
   255→
   256→  const handleAddRetour = async () => {
   257→    if (!pharmacie?.id || !retMedicamentId || !retQuantite) {
   258→      toast.error('Médicament et quantité sont requis')
   259→      return
   260→    }
   261→    try {
   262→      const res = await fetch('/api/retours', {
   263→        method: 'POST',
   264→        headers: { 'Content-Type': 'application/json' },
   265→        body: JSON.stringify({
   266→          pharmacieId: pharmacie.id,
   267→          medicamentId: retMedicamentId,
   268→          lotId: retLotId || null,
   269→          typeRetour: retType,
   270→          quantite: parseInt(retQuantite) || 0,
   271→          motif: retMotif,
   272→          fournisseurId: retFournisseurId || null,
   273→          statut: 'EN_COURS',
   274→        }),
   275→      })
   276→      if (res.ok) {
   277→        toast.success('Retour enregistré avec succès')
   278→        setAddRetourDialogOpen(false)
   279→        resetRetourForm()
   280→        fetchAllData()
   281→      } else {
   282→        toast.error("Erreur lors de l'enregistrement du retour")
   283→      }
   284→    } catch {
   285→      toast.error("Erreur lors de l'enregistrement du retour")
   286→    }
   287→  }
   288→
   289→  const handleAddDestruction = async () => {
   290→    if (!pharmacie?.id || !destMedicamentId || !destLotId || !destQuantite) {
   291→      toast.error('Médicament, lot et quantité sont requis')
   292→      return
   293→    }
   294→    try {
   295→      const res = await fetch('/api/destructions', {
   296→        method: 'POST',
   297→        headers: { 'Content-Type': 'application/json' },
   298→        body: JSON.stringify({
   299→          pharmacieId: pharmacie.id,
   300→          medicamentId: destMedicamentId,
   301→          lotId: destLotId,
   302→          quantite: parseInt(destQuantite) || 0,
   303→          motif: destMotif,
   304→          dateDestruction: destDate,
   305→          pharmacienId: destPharmacienId,
   306→          temoinId: destTemoinId || null,
   307→          statut: 'PLANIFIEE',
   308→        }),
   309→      })
   310→      if (res.ok) {
   311→        toast.success('Destruction planifiée avec succès')
   312→        setAddDestructionDialogOpen(false)
   313→        resetDestructionForm()
   314→        fetchAllData()
   315→      } else {
   316→        toast.error("Erreur lors de la planification de la destruction")
   317→      }
   318→    } catch {
   319→      toast.error("Erreur lors de la planification de la destruction")
   320→    }
   321→  }
   322→
   323→  const handleTraiterRetour = async () => {
   324→    if (!selectedRetour) return
   325→    try {
   326→      const res = await fetch('/api/retours', {
   327→        method: 'POST',
   328→        headers: { 'Content-Type': 'application/json' },
   329→        body: JSON.stringify({
   330→          id: selectedRetour.id,
   331→          pharmacieId: pharmacie?.id,
   332→          medicamentId: selectedRetour.medicamentId,
   333→          lotId: selectedRetour.lotId,
   334→          typeRetour: selectedRetour.typeRetour,
   335→          quantite: selectedRetour.quantite,
   336→          motif: selectedRetour.motif,
   337→          fournisseurId: selectedRetour.fournisseurId,
   338→          statut: 'TRAITE',
   339→        }),
   340→      })
   341→      if (res.ok) {
   342→        toast.success('Retour marqué comme traité')
   343→        setTraiterRetourDialogOpen(false)
   344→        setSelectedRetour(null)
   345→        fetchAllData()
   346→      } else {
   347→        toast.error('Erreur lors du traitement du retour')
   348→      }
   349→    } catch {
   350→      toast.error('Erreur lors du traitement du retour')
   351→    }
   352→  }
   353→
   354→  const handleUpdateDestructionStatut = async (newStatut: string) => {
   355→    if (!selectedDestruction || !pharmacie?.id) return
   356→    try {
   357→      const res = await fetch('/api/destructions', {
   358→        method: 'POST',
   359→        headers: { 'Content-Type': 'application/json' },
   360→        body: JSON.stringify({
   361→          id: selectedDestruction.id,
   362→          pharmacieId: pharmacie.id,
   363→          medicamentId: selectedDestruction.medicamentId,
   364→          lotId: selectedDestruction.lotId,
   365→          quantite: selectedDestruction.quantite,
   366→          motif: selectedDestruction.motif,
   367→          dateDestruction: selectedDestruction.dateDestruction,
   368→          pharmacienId: selectedDestruction.pharmacienId,
   369→          temoinId: selectedDestruction.temoinId,
   370→          statut: newStatut,
   371→        }),
   372→      })
   373→      if (res.ok) {
   374→        toast.success('Statut de destruction mis à jour')
   375→        setUpdateDestructionDialogOpen(false)
   376→        setSelectedDestruction(null)
   377→        fetchAllData()
   378→      } else {
   379→        toast.error('Erreur lors de la mise à jour')
   380→      }
   381→    } catch {
   382→      toast.error('Erreur lors de la mise à jour')
   383→    }
   384→  }
   385→
   386→  const resetRetourForm = () => {
   387→    setRetMedicamentId(''); setRetLotId(''); setRetType('PERIME'); setRetQuantite(''); setRetMotif(''); setRetFournisseurId('')
   388→  }
   389→  const resetDestructionForm = () => {
   390→    setDestMedicamentId(''); setDestLotId(''); setDestQuantite(''); setDestMotif(''); setDestDate(new Date().toISOString().slice(0, 10)); setDestTemoinId('')
   391→  }
   392→
   393→  if (loading) {
   394→    return (
   395→      <div className="space-y-4">
   396→        <Skeleton className="h-8 w-64" />
   397→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   398→          {[1, 2, 3, 4].map(i => <Skeleton key={i} className="h-24" />)}
   399→        </div>
   400→        <Skeleton className="h-96" />
   401→      </div>
   402→    )
   403→  }
   404→
   405→  return (
   406→    <div className="space-y-6">
   407→      {/* Header */}
   408→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   409→        <div>
   410→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   411→            <RotateCcw className="w-6 h-6 text-[#1D9E75]" />
   412→            Retours & Destructions
   413→          </h1>
   414→          <p className="text-sm text-muted-foreground">
   415→            Gestion des retours fournisseurs et destructions de médicaments
   416→          </p>
   417→        </div>
   418→      </div>
   419→
   420→      {/* KPI Cards */}
   421→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   422→        <Card>
   423→          <CardContent className="p-4">
   424→            <div className="flex items-center justify-between">
   425→              <div>
   426→                <span className="text-xs text-muted-foreground uppercase">Retours ce mois</span>
   427→                <span className="text-xl font-bold block">{retoursCeMois}</span>
   428→              </div>
   429→              <Calendar className="w-8 h-8 text-gray-400/30" />
   430→            </div>
   431→          </CardContent>
   432→        </Card>
   433→        <Card>
   434→          <CardContent className="p-4">
   435→            <div className="flex items-center justify-between">
   436→              <div>
   437→                <span className="text-xs text-muted-foreground uppercase">Valeur retours</span>
   438→                <span className="text-lg font-bold block text-amber-600">{formatFCFA(valeurRetours)}</span>
   439→              </div>
   440→              <Package className="w-8 h-8 text-amber-400/30" />
   441→            </div>
   442→          </CardContent>
   443→        </Card>
   444→        <Card>
   445→          <CardContent className="p-4">
   446→            <div className="flex items-center justify-between">
   447→              <div>
   448→                <span className="text-xs text-muted-foreground uppercase">En attente</span>
   449→                <span className="text-xl font-bold block text-red-600">{enAttente}</span>
   450→              </div>
   451→              <Clock className="w-8 h-8 text-red-400/30" />
   452→            </div>
   453→          </CardContent>
   454→        </Card>
   455→        <Card>
   456→          <CardContent className="p-4">
   457→            <div className="flex items-center justify-between">
   458→              <div>
   459→                <span className="text-xs text-muted-foreground uppercase">Traités</span>
   460→                <span className="text-xl font-bold block text-[#1D9E75]">{traites}</span>
   461→              </div>
   462→              <CheckCircle2 className="w-8 h-8 text-[#1D9E75]/30" />
   463→            </div>
   464→          </CardContent>
   465→        </Card>
   466→      </div>
   467→
   468→      {/* Tabs */}
   469→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   470→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   471→          <TabsList>
   472→            <TabsTrigger value="retours">Retours</TabsTrigger>
   473→            <TabsTrigger value="destructions">Destructions</TabsTrigger>
   474→          </TabsList>
   475→          <div className="flex items-center gap-2">
   476→            <div className="relative flex-1 min-w-[200px]">
   477→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   478→              <Input
   479→                placeholder="Rechercher..."
   480→                className="pl-9"
   481→                value={search}
   482→                onChange={e => setSearch(e.target.value)}
   483→              />
   484→            </div>
   485→            {activeTab === 'retours' && (
   486→              <Button className="gap-2 bg-[#1D9E75] hover:bg-[#085041]" onClick={() => { resetRetourForm(); setAddRetourDialogOpen(true) }}>
   487→                <Plus className="w-4 h-4" /> Retour
   488→              </Button>
   489→            )}
   490→            {activeTab === 'destructions' && (
   491→              <Button className="gap-2 bg-[#1D9E75] hover:bg-[#085041]" onClick={() => { resetDestructionForm(); setAddDestructionDialogOpen(true) }}>
   492→                <Plus className="w-4 h-4" /> Destruction
   493→              </Button>
   494→            )}
   495→          </div>
   496→        </div>
   497→
   498→        {/* Retours Tab */}
   499→        <TabsContent value="retours" className="mt-4">
   500→          {/* Filters */}
   501→          <Card className="mb-4">
   502→            <CardContent className="p-4">
   503→              <div className="flex flex-col sm:flex-row gap-3">
   504→                <Select value={filterRetourType} onValueChange={setFilterRetourType}>
   505→                  <SelectTrigger className="w-48">
   506→                    <SelectValue />
   507→                  </SelectTrigger>
   508→                  <SelectContent>
