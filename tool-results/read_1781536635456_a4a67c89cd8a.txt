   127→function formatFCFA(amount: number): string {
   128→  return new Intl.NumberFormat('fr-FR', {
   129→    style: 'decimal',
   130→    minimumFractionDigits: 0,
   131→    maximumFractionDigits: 0,
   132→  }).format(amount) + ' FCFA'
   133→}
   134→
   135→function formatDate(date: string | null): string {
   136→  if (!date) return '—'
   137→  return new Date(date).toLocaleDateString('fr-FR', {
   138→    day: '2-digit',
   139→    month: '2-digit',
   140→    year: 'numeric',
   141→  })
   142→}
   143→
   144→function formatDateLong(date: string): string {
   145→  return new Date(date).toLocaleDateString('fr-FR', {
   146→    day: 'numeric',
   147→    month: 'long',
   148→    year: 'numeric',
   149→  })
   150→}

   208→export default function PatientsPage() {
   209→  const { pharmacie } = useAuth()
   210→  const pharmacieId = pharmacie?.id || ''
   211→
   212→  // --- State ---
   213→  const [patients, setPatients] = useState<Patient[]>([])
   214→  const [total, setTotal] = useState(0)
   215→  const [totalPages, setTotalPages] = useState(1)
   216→  const [stats, setStats] = useState(EMPTY_STATS)
   217→  const [loading, setLoading] = useState(true)
   218→
   219→  // Filters
   220→  const [search, setSearch] = useState('')
   221→  const [actifFilter, setActifFilter] = useState<string>('all')
   222→  const [assuranceFilter, setAssuranceFilter] = useState<string>('all')
   223→  const [creditFilter, setCreditFilter] = useState<string>('all')
   224→  const [sortBy, setSortBy] = useState<string>('nom')
   225→  const [sortOrder, setSortOrder] = useState<string>('asc')
   226→  const [page, setPage] = useState(1)
   227→  const limit = 20
   228→
   229→  // Dialogs
   230→  const [showAddDialog, setShowAddDialog] = useState(false)
   231→  const [showEditDialog, setShowEditDialog] = useState(false)
   232→  const [showDetailDialog, setShowDetailDialog] = useState(false)
   233→  const [showCreditDialog, setShowCreditDialog] = useState(false)
   234→  const [selectedPatient, setSelectedPatient] = useState<PatientDetail | null>(null)
   235→  const [editPatient, setEditPatient] = useState<Patient | null>(null)
   236→  const [creditPatient, setCreditPatient] = useState<Patient | null>(null)
   237→  const [detailLoading, setDetailLoading] = useState(false)
   238→  const [saving, setSaving] = useState(false)
   239→
   240→  // Form state
   241→  const emptyForm = {
   242→    nom: '', prenom: '', telephone: '', email: '', dateNaissance: '',
   243→    sexe: '', numeroAssurance: '', assurance: 'none', adresse: '', notes: '',
   244→    creditAutorise: false, creditLimite: 0,
   245→  }
   246→  const [form, setForm] = useState(emptyForm)
   247→
   248→  // --- Fetch patients ---
   249→  const fetchPatients = useCallback(async () => {
   250→    if (!pharmacieId) return
   251→    setLoading(true)
   252→    try {
   253→      const params = new URLSearchParams({
   254→        pharmacieId,
   255→        page: String(page),
   256→        limit: String(limit),
   257→        sortBy,
   258→        sortOrder,
   259→      })
   260→      if (search) params.set('search', search)
   261→      if (actifFilter !== 'all') params.set('actif', actifFilter)
   262→      if (assuranceFilter !== 'all') params.set('assurance', assuranceFilter)
   263→      if (creditFilter !== 'all') params.set('creditStatus', creditFilter)
   264→
   265→      const res = await fetch(`/api/patients?${params}`)
   266→      if (!res.ok) throw new Error()
   267→      const data: PatientsResponse = await res.json()
   268→      setPatients(data.patients)
   269→      setTotal(data.total)
   270→      setTotalPages(data.totalPages)
   271→      setStats(data.stats)
   272→    } catch {
   273→      toast.error('Erreur lors du chargement des patients')
   274→    } finally {
   275→      setLoading(false)
   276→    }
   277→  }, [pharmacieId, page, sortBy, sortOrder, search, actifFilter, assuranceFilter, creditFilter])
   278→
   279→  useEffect(() => {
   280→    fetchPatients()
   281→  }, [fetchPatients])
   282→
   283→  // Reset to page 1 when filters change
   284→  useEffect(() => {
   285→    setPage(1)
   286→  }, [search, actifFilter, assuranceFilter, creditFilter, sortBy, sortOrder])
   287→

   289→  const fetchPatientDetail = async (patientId: string) => {
   290→    if (!pharmacieId) return
   291→    setDetailLoading(true)
   292→    try {
   293→      const res = await fetch(`/api/patients/${patientId}?pharmacieId=${pharmacieId}`)
   294→      if (!res.ok) throw new Error()
   295→      const data: PatientDetail = await res.json()
   296→      setSelectedPatient(data)
   297→    } catch {
   298→      toast.error('Erreur lors du chargement du patient')
   299→    } finally {
   300→      setDetailLoading(false)
   301→    }
   302→  }
   303→
   304→  // --- Save patient (add/edit) ---
   305→  const handleSavePatient = async (isEdit: boolean) => {
   306→    if (!pharmacieId || !form.nom || !form.prenom) {
   307→      toast.error('Nom et prénom sont requis')
   308→      return
   309→    }
   310→    setSaving(true)
   311→    try {
   312→      const payload = {
   313→        pharmacieId,
   314→        nom: form.nom,
   315→        prenom: form.prenom,
   316→        telephone: form.telephone,
   317→        email: form.email || null,
   318→        dateNaissance: form.dateNaissance || null,
   319→        sexe: form.sexe || null,
   320→        numeroAssurance: form.numeroAssurance || null,
   321→        assurance: form.assurance === 'none' ? null : form.assurance,
   322→        adresse: form.adresse || null,
   323→        notes: form.notes || null,
   324→        creditAutorise: form.creditAutorise,
   325→        creditLimite: form.creditAutorise ? form.creditLimite : 0,
   326→      }
   327→
   328→      if (isEdit && editPatient) {
   329→        const res = await fetch(`/api/patients/${editPatient.id}?pharmacieId=${pharmacieId}`, {
   330→          method: 'PUT',
   331→          headers: { 'Content-Type': 'application/json' },
   332→          body: JSON.stringify(payload),
   333→        })
   334→        if (!res.ok) throw new Error()
   335→        toast.success('Patient mis à jour avec succès')
   336→        setShowEditDialog(false)
   337→      } else {
   338→        const res = await fetch('/api/patients', {
   339→          method: 'POST',
   340→          headers: { 'Content-Type': 'application/json' },
   341→          body: JSON.stringify(payload),
   342→        })
   343→        if (!res.ok) throw new Error()
   344→        toast.success('Patient créé avec succès')
   345→        setShowAddDialog(false)
   346→      }
   347→      setForm(emptyForm)
   348→      setEditPatient(null)
   349→      fetchPatients()
   350→    } catch {
   351→      toast.error(isEdit ? 'Erreur lors de la mise à jour' : 'Erreur lors de la création')
   352→    } finally {
   353→      setSaving(false)
   354→    }
   355→  }
   356→
   357→  // --- Toggle credit ---
   358→  const handleSaveCredit = async () => {
   359→    if (!pharmacieId || !creditPatient) return
   360→    setSaving(true)
   361→    try {
   362→      const res = await fetch(`/api/patients/${creditPatient.id}?pharmacieId=${pharmacieId}`, {
   363→        method: 'PATCH',
   364→        headers: { 'Content-Type': 'application/json' },
   365→        body: JSON.stringify({
   366→          pharmacieId,
   367→          creditAutorise: creditPatient.creditAutorise,
   368→          creditLimite: creditPatient.creditLimite,
   369→        }),
   370→      })
   371→      if (!res.ok) throw new Error()
   372→      toast.success('Crédit mis à jour')
   373→      setShowCreditDialog(false)
   374→      setCreditPatient(null)
   375→      fetchPatients()
   376→    } catch {
   377→      toast.error('Erreur lors de la mise à jour du crédit')
   378→    } finally {
   379→      setSaving(false)
   380→    }
   381→  }
   382→
   383→  // --- Toggle active ---
   384→  const handleToggleActive = async (patient: Patient) => {
   385→    if (!pharmacieId) return
   386→    try {
   387→      const res = await fetch(`/api/patients/${patient.id}?pharmacieId=${pharmacieId}`, {
   388→        method: 'PATCH',
   389→        headers: { 'Content-Type': 'application/json' },
   390→        body: JSON.stringify({
   391→          pharmacieId,
   392→          actif: !patient.actif,
   393→        }),
   394→      })
   395→      if (!res.ok) throw new Error()
   396→      toast.success(patient.actif ? 'Patient désactivé' : 'Patient réactivé')
   397→      fetchPatients()
   398→    } catch {
   399→      toast.error('Erreur lors du changement de statut')
   400→    }
   401→  }
   402→
   403→  // --- Open edit dialog ---
   404→  const openEdit = (patient: Patient) => {
   405→    setEditPatient(patient)
   406→    setForm({
   407→      nom: patient.nom,
   408→      prenom: patient.prenom,
   409→      telephone: patient.telephone,
   410→      email: patient.email || '',
   411→      dateNaissance: patient.dateNaissance
   412→        ? new Date(patient.dateNaissance).toISOString().split('T')[0]
   413→        : '',
   414→      sexe: patient.sexe || '',
   415→      numeroAssurance: patient.numeroAssurance || '',
   416→      assurance: patient.assurance || 'none',
   417→      adresse: patient.adresse || '',
   418→      notes: patient.notes || '',
   419→      creditAutorise: patient.creditAutorise,
   420→      creditLimite: patient.creditLimite,
   421→    })
   422→    setShowEditDialog(true)
   423→  }
   424→
   425→  // --- Open detail dialog ---
   426→  const openDetail = (patient: Patient) => {
   427→    fetchPatientDetail(patient.id)
   428→    setShowDetailDialog(true)
   429→  }
   430→
   431→  // --- Sort toggle ---
   432→  const handleSort = (field: string) => {
   433→    if (sortBy === field) {
   434→      setSortOrder(sortOrder === 'asc' ? 'desc' : 'asc')
   435→    } else {
   436→      setSortBy(field)
   437→      setSortOrder('asc')
   438→    }
   439→  }
   440→
   441→  // --- Memoized filtered check ---
   442→  const hasActiveFilters = useMemo(() => {
   443→    return search !== '' || actifFilter !== 'all' || assuranceFilter !== 'all' || creditFilter !== 'all'
   444→  }, [search, actifFilter, assuranceFilter, creditFilter])
   445→
   446→  // ============================================================
   447→  // Render
   448→  // ============================================================
   449→
   450→  return (
   451→    <div className="space-y-6">
   452→      {/* Header */}
   453→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   454→        <div>
   455→          <h1 className="text-2xl font-bold tracking-tight">Patients</h1>
   456→          <p className="text-sm text-muted-foreground">
   457→            Gérez vos patients, leurs informations et crédits
   458→          </p>
   459→        </div>
   460→        <Button
   461→          onClick={() => {
   462→            setForm(emptyForm)
   463→            setShowAddDialog(true)
   464→          }}
   465→          className="bg-[#1D9E75] hover:bg-[#1D9E75]/90 text-white gap-2"
   466→        >
   467→          <UserPlus className="w-4 h-4" />
   468→          Nouveau patient
   469→        </Button>
   470→      </div>
   471→
   472→      {/* KPI Cards */}
   473→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   474→        <KpiCard
   475→          title="Patients actifs"
   476→          value={stats.totalActifs}
   477→          icon={Users}
   478→          variant="success"
   479→        />
   480→        <KpiCard
   481→          title="Avec assurance"
   482→          value={stats.patientsAvecAssurance}
   483→          icon={Shield}
   484→          variant="default"
   485→        />
   486→        <KpiCard
   487→          title="Crédits en cours"
   488→          value={formatFCFA(stats.creditsEnCours)}
   489→          icon={CreditCard}
   490→          variant="warning"
   491→        />
   492→        <KpiCard
   493→          title="Nouveaux ce mois"
   494→          value={stats.nouveauxMois}
   495→          icon={Activity}
   496→          variant="default"
   497→        />
   498→      </div>
   499→
   500→      {/* Filters Bar */}
   501→      <Card>
   502→        <CardContent className="p-4">
   503→          <div className="flex flex-col lg:flex-row gap-3">
   504→            {/* Search */}
   505→            <div className="relative flex-1">
   506→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   507→              <Input
   508→                placeholder="Rechercher par nom, téléphone, email..."
   509→                value={search}
   510→                onChange={(e) => setSearch(e.target.value)}
   511→                className="pl-9"
   512→              />
   513→              {search && (
   514→                <button
   515→                  onClick={() => setSearch('')}
   516→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   517→                >
   518→                  <X className="w-4 h-4" />
   519→                </button>
   520→              )}
   521→            </div>
   522→
   523→            {/* Filters */}
   524→            <div className="flex flex-wrap gap-2">
   525→              <Select value={actifFilter} onValueChange={setActifFilter}>
   526→                <SelectTrigger className="w-[150px]">
   527→                  <SelectValue placeholder="Statut" />
   528→                </SelectTrigger>
   529→                <SelectContent>
   530→                  <SelectItem value="all">Tous les statuts</SelectItem>
   531→                  <SelectItem value="true">Actif</SelectItem>
   532→                  <SelectItem value="false">Inactif</SelectItem>
   533→                </SelectContent>
   534→              </Select>
   535→
   536→              <Select value={assuranceFilter} onValueChange={setAssuranceFilter}>
   537→                <SelectTrigger className="w-[170px]">
   538→                  <SelectValue placeholder="Assurance" />
   539→                </SelectTrigger>
   540→                <SelectContent>
   541→                  <SelectItem value="all">Toutes assurances</SelectItem>
   542→                  <SelectItem value="CNSS">CNSS</SelectItem>
   543→                  <SelectItem value="RAMU">RAMU</SelectItem>
   544→                  <SelectItem value="ASSURANCE_PRIVEE">Assurance privée</SelectItem>
   545→                </SelectContent>
   546→              </Select>
   547→
   548→              <Select value={creditFilter} onValueChange={setCreditFilter}>
   549→                <SelectTrigger className="w-[160px]">
   550→                  <SelectValue placeholder="Crédit" />
   551→                </SelectTrigger>
   552→                <SelectContent>
   553→                  <SelectItem value="all">Tous crédits</SelectItem>
   554→                  <SelectItem value="autorise">Crédit autorisé</SelectItem>
   555→                  <SelectItem value="non_autorise">Sans crédit</SelectItem>
   556→                </SelectContent>
   557→              </Select>
   558→
   559→              {hasActiveFilters && (
   560→                <Button
   561→                  variant="ghost"
   562→                  size="sm"
   563→                  onClick={() => {
   564→                    setSearch('')
   565→                    setActifFilter('all')
   566→                    setAssuranceFilter('all')
   567→                    setCreditFilter('all')
   568→                  }}
   569→                  className="gap-1 text-muted-foreground"
   570→                >
   571→                  <X className="w-3 h-3" />
   572→                  Réinitialiser
   573→                </Button>
   574→              )}
   575→            </div>
   576→          </div>
   577→        </CardContent>
   578→      </Card>
   579→
   580→      {/* Table */}
   581→      <Card>
   582→        <CardContent className="p-0">
   583→          {loading ? (
   584→            <div className="flex items-center justify-center py-20">
   585→              <Loader2 className="w-8 h-8 animate-spin text-muted-foreground" />
   586→            </div>
   587→          ) : patients.length === 0 ? (
   588→            <div className="flex flex-col items-center justify-center py-20 text-center">
   589→              <Users className="w-12 h-12 text-muted-foreground/40 mb-4" />
   590→              <p className="text-lg font-medium text-muted-foreground">Aucun patient trouvé</p>
   591→              <p className="text-sm text-muted-foreground/70 mt-1">
   592→                {hasActiveFilters
   593→                  ? 'Essayez de modifier vos filtres de recherche'
   594→                  : 'Ajoutez votre premier patient pour commencer'}
   595→              </p>
   596→              {!hasActiveFilters && (
   597→                <Button
   598→                  onClick={() => {
   599→                    setForm(emptyForm)
   600→                    setShowAddDialog(true)
   601→                  }}
   602→                  className="mt-4 bg-[#1D9E75] hover:bg-[#1D9E75]/90 text-white gap-2"
   603→                >
   604→                  <UserPlus className="w-4 h-4" />
   605→                  Ajouter un patient
