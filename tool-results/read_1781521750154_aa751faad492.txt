    91→export default function VentesPage() {
    92→  const { pharmacie } = useAuth()
    93→  const [medicaments, setMedicaments] = useState<Medicament[]>([])
    94→  const [ventes, setVentes] = useState<unknown[]>([])
    95→  const [loading, setLoading] = useState(true)
    96→  const [search, setSearch] = useState('')
    97→  const [cart, setCart] = useState<CartItem[]>([])
    98→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
    99→  const [montantPaye, setMontantPaye] = useState('')
   100→  const [activeTab, setActiveTab] = useState<'pos' | 'historique'>('pos')
   101→
   102→  // Session caisse
   103→  const [sessionCaisse, setSessionCaisse] = useState<SessionCaisse | null>(null)
   104→  const [openSessionDialog, setOpenSessionDialog] = useState(false)
   105→  const [closeSessionDialog, setCloseSessionDialog] = useState(false)
   106→  const [fondDeCaisse, setFondDeCaisse] = useState('0')
   107→  const [soldePhysique, setSoldePhysique] = useState('')
   108→  const [zReport, setZReport] = useState<Record<string, unknown> | null>(null)
   109→
   110→  // Patient & Ordonnance
   111→  const [patients, setPatients] = useState<Patient[]>([])
   112→  const [ordonnances, setOrdonnances] = useState<Ordonnance[]>([])
   113→  const [selectedPatientId, setSelectedPatientId] = useState<string>('')
   114→  const [selectedOrdonnanceId, setSelectedOrdonnanceId] = useState<string>('')
   115→
   116→  // Remise globale
   117→  const [remiseGlobale, setRemiseGlobale] = useState(0)
   118→
   119→  // Multi-paiement
   120→  const [multiPaiements, setMultiPaiements] = useState<MultiPayment[]>([
   121→    { id: '1', modePaiement: 'ESPECES', montant: 0 }
   122→  ])
   123→
   124→  // Receipt
   125→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   126→  const [lastVente, setLastVente] = useState<Record<string, unknown> | null>(null)
   127→  const [receiptCart, setReceiptCart] = useState<CartItem[]>([])
   128→  const [receiptRemise, setReceiptRemise] = useState(0)
   129→  const [receiptTVA, setReceiptTVA] = useState(0)
   130→  const [receiptTotal, setReceiptTotal] = useState(0)
   131→  const [receiptPatient, setReceiptPatient] = useState<string>('')
   132→  const [receiptPaiements, setReceiptPaiements] = useState<MultiPayment[]>([])
   133→
   134→  // Stupéfiant warning
   135→  const [stupWarningOpen, setStupWarningOpen] = useState(false)
   136→
   137→  // Ticket dialog for historique
   138→  const [ticketDialogOpen, setTicketDialogOpen] = useState(false)
   139→  const [ticketData, setTicketData] = useState<Record<string, unknown> | null>(null)
   140→  const [ticketLoading, setTicketLoading] = useState(false)
   141→
   142→  useEffect(() => {
   143→    const pid = pharmacie?.id
   144→    if (!pid) return
   145→    Promise.all([
   146→      fetch(`/api/medicaments?pharmacieId=${pid}`).then(r => r.ok ? r.json() : []),
   147→      fetch(`/api/ventes?pharmacieId=${pid}`).then(r => r.ok ? r.json() : []),
   148→      fetch(`/api/patients?pharmacieId=${pid}`).then(r => r.ok ? r.json() : []),
   149→      fetch(`/api/ordonnances?pharmacieId=${pid}&statut=VALIDEE`).then(r => r.ok ? r.json() : []),
   150→      fetch(`/api/sessions-caisse?pharmacieId=${pid}&statut=ouverte`).then(r => r.ok ? r.json() : []),
   151→    ]).then(([meds, vents, pats, ordos, sessions]) => {
   152→      setMedicaments(meds)
   153→      setVentes(vents)
   154→      setPatients(pats)
   155→      setOrdonnances(ordos)
   156→      setSessionCaisse(sessions.length > 0 ? sessions[0] : null)
   157→      setLoading(false)
   158→    }).catch(() => { setLoading(false) })
   159→  }, [pharmacie?.id])
   160→
   161→  const filteredMeds = useMemo(() => {
   162→    if (!search) return medicaments.slice(0, 20)
   163→    const q = search.toLowerCase()
   164→    return medicaments.filter(m =>
   165→      m.nomCommercial.toLowerCase().includes(q) ||
   166→      m.dci.toLowerCase().includes(q)
   167→    ).slice(0, 20)
   168→  }, [medicaments, search])
   169→
   170→  const addToCart = (med: Medicament) => {
   171→    setCart(prev => {
   172→      const existing = prev.find(c => c.medicament.id === med.id)
   173→      if (existing) {
   174→        return prev.map(c =>
   175→          c.medicament.id === med.id
   176→            ? { ...c, quantite: c.quantite + 1 }
   177→            : c
   178→        )
   179→      }
   180→      return [...prev, { medicament: med, quantite: 1, lotId: med.lots[0]?.id || null, remise: 0 }]
   181→    })
   182→  }
   183→
   184→  const updateQuantity = (medId: string, delta: number) => {
   185→    setCart(prev => {
   186→      const updated = prev.map(c => {
   187→        if (c.medicament.id === medId) {
   188→          return { ...c, quantite: c.quantite + delta }
   189→        }
   190→        return c
   191→      })
   192→      return updated.filter(c => c.quantite > 0)
   193→    })
   194→  }
   195→
   196→  const updateRemise = (medId: string, remise: number) => {
   197→    setCart(prev => prev.map(c =>
   198→      c.medicament.id === medId ? { ...c, remise: Math.min(remise, 100) } : c
   199→    ))
   200→  }
   201→
   202→  const removeFromCart = (medId: string) => {
   203→    setCart(prev => prev.filter(c => c.medicament.id !== medId))
   204→  }
   205→
   206→  // Calculs
   207→  const sousTotal = cart.reduce((sum, c) => {
   208→    const lineTotal = c.medicament.prixVente * c.quantite
   209→    const lineRemise = lineTotal * (c.remise / 100)
   210→    return sum + lineTotal - lineRemise
   211→  }, 0)
   212→  const remiseGlobaleMontant = sousTotal * (remiseGlobale / 100)
   213→  const totalApresRemise = sousTotal - remiseGlobaleMontant
   214→
   215→  // TVA
   216→  const totalTVA = cart.reduce((sum, c) => {
   217→    const lineTotal = c.medicament.prixVente * c.quantite
   218→    const lineRemise = lineTotal * (c.remise / 100)
   219→    const netLine = lineTotal - lineRemise
   220→    return sum + (netLine * c.medicament.tva / 100)
   221→  }, 0)
   222→
   223→  const totalCart = totalApresRemise
   224→
   225→  // Stupéfiant check
   226→  const hasStupfiant = cart.some(c => c.medicament.estStupefiant)
   227→  const stupfiantRequiresOrdo = hasStupfiant && !selectedOrdonnanceId
   228→
   229→  // Multi-paiement
   230→  const totalPaiements = multiPaiements.reduce((s, p) => s + p.montant, 0)
   231→  const paiementRestant = totalCart - totalPaiements
   232→  const monnaie = paiementRestant < 0 ? Math.abs(paiementRestant) : 0
   233→
   234→  const addPaiement = () => {
   235→    setMultiPaiements(prev => [...prev, { id: Date.now().toString(), modePaiement: 'ESPECES', montant: 0 }])
   236→  }
   237→
   238→  const removePaiement = (id: string) => {
   239→    setMultiPaiements(prev => prev.filter(p => p.id !== id))
   240→  }
   241→
   242→  const updatePaiement = (id: string, field: 'modePaiement' | 'montant', value: string | number) => {
   243→    setMultiPaiements(prev => prev.map(p =>
   244→      p.id === id ? { ...p, [field]: value } : p
   245→    ))
   246→  }
   247→
   248→  // Ouvrir session caisse
   249→  const handleOpenSession = async () => {
   250→    if (!pharmacie?.id) return
   251→    try {
   252→      // Trouver une caisse disponible pour cette pharmacie
   253→      const caissesRes = await fetch(`/api/caisses?pharmacieId=${pharmacie.id}`)
   254→      let caisseId = 'demo-caisse'
   255→      if (caissesRes.ok) {
   256→        const caissesData = await caissesRes.json()
   257→        if (caissesData.length > 0) {
   258→          // Prendre la première caisse sans session ouverte
   259→          const availableCaisse = caissesData.find(
   260→            (c: { id: string; sessionsCaisse: { id: string }[] }) => !c.sessionsCaisse || c.sessionsCaisse.length === 0
   261→          )
   262→          caisseId = availableCaisse?.id || caissesData[0]?.id || 'demo-caisse'
   263→        } else {
   264→          // Créer une caisse par défaut
   265→          const createRes = await fetch('/api/caisses', {
   266→            method: 'POST',
   267→            headers: { 'Content-Type': 'application/json' },
   268→            body: JSON.stringify({
   269→              pharmacieId: pharmacie.id,
   270→              nom: 'Caisse Principale',
   271→              numero: 1,
   272→            }),
   273→          })
   274→          if (createRes.ok) {
   275→            const newCaisse = await createRes.json()
   276→            caisseId = newCaisse.id
   277→          }
   278→        }
   279→      }
   280→
   281→      const res = await fetch('/api/sessions-caisse', {
   282→        method: 'POST',
   283→        headers: { 'Content-Type': 'application/json' },
   284→        body: JSON.stringify({
   285→          pharmacieId: pharmacie.id,
   286→          utilisateurId: 'demo-admin',
   287→          caisseId,
   288→          fondDeCaisse: parseFloat(fondDeCaisse) || 0,
   289→        }),
   290→      })
   291→
   292→      if (res.ok) {
   293→        const data = await res.json()
   294→        setSessionCaisse(data)
   295→        setOpenSessionDialog(false)
   296→        toast.success('Session de caisse ouverte')
   297→      } else {
   298→        const err = await res.json()
   299→        toast.error(err.error || 'Erreur ouverture session')
   300→      }
   301→    } catch {
   302→      toast.error('Erreur ouverture session')
   303→    }
   304→  }
   305→
   306→  // Fermer session caisse
   307→  const handleCloseSession = async () => {
   308→    if (!sessionCaisse) return
   309→    try {
   310→      const res = await fetch(`/api/sessions-caisse/${sessionCaisse.id}`, {
   311→        method: 'PATCH',
   312→        headers: { 'Content-Type': 'application/json' },
   313→        body: JSON.stringify({
   314→          soldePhysique: soldePhysique ? parseFloat(soldePhysique) : undefined,
   315→          ecart: soldePhysique ? undefined : 0,
   316→        }),
   317→      })
   318→      if (res.ok) {
   319→        const data = await res.json()
   320→        setZReport(data.zReport)
   321→        setSessionCaisse(null)
   322→        setCloseSessionDialog(false)
   323→        setSoldePhysique('')
   324→        toast.success('Session fermée — Z-report généré')
   325→      }
   326→    } catch {
   327→      toast.error('Erreur fermeture session')
   328→    }
   329→  }
   330→
   331→  // Valider vente
   332→  const handleValidateSale = async () => {
   333→    if (!pharmacie?.id || cart.length === 0) return
   334→
   335→    if (stupfiantRequiresOrdo) {
   336→      setStupWarningOpen(true)
   337→      return
   338→    }
   339→
   340→    try {
   341→      const lignes = cart.map(c => ({
   342→        medicamentId: c.medicament.id,
   343→        lotId: c.lotId,
   344→        quantite: c.quantite,
   345→        prixUnitaire: c.medicament.prixVente,
   346→        montant: c.medicament.prixVente * c.quantite * (1 - c.remise / 100),
   347→        remise: c.remise,
   348→      }))
   349→
   350→      const paiementsToSend = multiPaiements
   351→        .filter(p => p.montant > 0)
   352→        .map(p => ({
   353→          modePaiement: p.modePaiement,
   354→          montant: p.montant,
   355→        }))
   356→
   357→      if (paiementsToSend.length === 0) {
   358→        paiementsToSend.push({
   359→          modePaiement: 'ESPECES',
   360→          montant: totalCart,
   361→        })
   362→      }
   363→
   364→      const res = await fetch('/api/ventes', {
   365→        method: 'POST',
   366→        headers: { 'Content-Type': 'application/json' },
   367→        body: JSON.stringify({
   368→          pharmacieId: pharmacie.id,
   369→          utilisateurId: 'demo-admin',
   370→          sessionId: sessionCaisse?.id || null,
   371→          patientId: selectedPatientId || null,
   372→          ordonnanceId: selectedOrdonnanceId || null,
   373→          typeVente: selectedOrdonnanceId ? 'ORDONNANCE' : 'COMPTOIR',
   374→          statut: 'VALIDEE',
   375→          montantTotal: totalCart,
   376→          montantRemise: remiseGlobaleMontant,
   377→          montantPaye: Math.min(totalPaiements, totalCart),
   378→          monnaieRendue: Math.max(0, monnaie),
   379→          lignes,
   380→          paiements: paiementsToSend,
   381→        }),
   382→      })
   383→
   384→      if (res.ok) {
   385→        const venteData = await res.json()
   386→        setLastVente(venteData)
   387→        // Sauvegarder les données du reçu avant de vider le panier
   388→        setReceiptCart([...cart])
   389→        setReceiptRemise(remiseGlobaleMontant)
   390→        setReceiptTVA(totalTVA)
   391→        setReceiptTotal(totalCart)
   392→        const selPatient = patients.find(p => p.id === selectedPatientId)
   393→        setReceiptPatient(selPatient ? `${selPatient.prenom} ${selPatient.nom}` : '')
   394→        setReceiptPaiements([...multiPaiements.filter(p => p.montant > 0)])
   395→        setCart([])
   396→        setPaymentDialogOpen(false)
   397→        setMontantPaye('')
   398→        setRemiseGlobale(0)
   399→        setSelectedPatientId('')
   400→        setSelectedOrdonnanceId('')
   401→        setMultiPaiements([{ id: '1', modePaiement: 'ESPECES', montant: 0 }])
   402→        // Afficher le reçu
   403→        setReceiptDialogOpen(true)
   404→        // Refresh ventes
   405→        const ventesRes = await fetch(`/api/ventes?pharmacieId=${pharmacie.id}`)
   406→        if (ventesRes.ok) setVentes(await ventesRes.json())
   407→        // Refresh session
   408→        const sessRes = await fetch(`/api/sessions-caisse?pharmacieId=${pharmacie.id}&statut=ouverte`)
   409→        if (sessRes.ok) {
   410→          const sessions = await sessRes.json()
   411→          setSessionCaisse(sessions.length > 0 ? sessions[0] : null)
   412→        }
   413→      }
   414→    } catch {
   415→      toast.error('Erreur lors de la validation')
   416→    }
   417→  }
   418→
   419→  if (loading) {
   420→    return (
   421→      <div className="space-y-4">
   422→        <Skeleton className="h-8 w-48" />
   423→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   424→          <Skeleton className="h-96 lg:col-span-2" />
   425→          <Skeleton className="h-96" />
   426→        </div>
   427→      </div>
   428→    )
   429→  }
   430→
   431→  return (
   432→    <div className="space-y-4">
   433→      {/* Header */}
   434→      <div className="flex items-center justify-between">
   435→        <div>
   436→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   437→            <ShoppingCart className="w-6 h-6 text-primary" />
   438→            Point de Vente
   439→          </h1>
   440→          <p className="text-sm text-muted-foreground">
   441→            Créez une vente et encaissez
   442→          </p>
   443→        </div>
   444→        <div className="flex gap-2">
   445→          {/* Session caisse */}
   446→          {sessionCaisse ? (
   447→            <div className="flex items-center gap-2">
   448→              <Badge className="bg-primary text-primary-foreground flex items-center gap-1">
   449→                <Unlock className="w-3 h-3" />
   450→                Caisse ouverte
   451→              </Badge>
   452→              <Button variant="outline" size="sm" onClick={() => setCloseSessionDialog(true)}>
   453→                <Lock className="w-3 h-3 mr-1" /> Fermer session
   454→              </Button>
   455→            </div>
   456→          ) : (
   457→            <Button variant="outline" size="sm" onClick={() => setOpenSessionDialog(true)}>
   458→              <Unlock className="w-3 h-3 mr-1" /> Ouvrir session
   459→            </Button>
   460→          )}
   461→          <Button
   462→            variant={activeTab === 'pos' ? 'default' : 'outline'}
   463→            size="sm"
   464→            onClick={() => setActiveTab('pos')}
   465→          >
   466→            Caisse
   467→          </Button>
   468→          <Button
   469→            variant={activeTab === 'historique' ? 'default' : 'outline'}
   470→            size="sm"
   471→            onClick={() => setActiveTab('historique')}
   472→          >
   473→            Historique
   474→          </Button>
   475→        </div>
   476→      </div>
   477→
   478→      {/* Session info bar */}
   479→      {sessionCaisse && (
   480→        <Card className="bg-primary/5 border-primary/20">
   481→          <CardContent className="p-3 flex items-center justify-between text-sm">
   482→            <div className="flex items-center gap-4">
   483→              <span className="text-muted-foreground">Caisse: <strong>{sessionCaisse.caisse?.nom || 'Principale'}</strong></span>
   484→              <span className="text-muted-foreground">Ouverte: <strong>{new Date(sessionCaisse.dateOuverture).toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })}</strong></span>
   485→              <span className="text-muted-foreground">Fond: <strong>{formatFCFA(sessionCaisse.fondDeCaisse)}</strong></span>
   486→              <span className="text-muted-foreground">Entrées: <strong className="text-primary">{formatFCFA(sessionCaisse.totalEntrees)}</strong></span>
   487→            </div>
   488→            <span className="text-muted-foreground">
   489→              Opérateur: <strong>{sessionCaisse.utilisateur?.prenom} {sessionCaisse.utilisateur?.nom}</strong>
   490→            </span>
   491→          </CardContent>
   492→        </Card>
   493→      )}
   494→
   495→      {activeTab === 'pos' ? (
   496→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   497→          {/* Product Search / List */}
   498→          <div className="lg:col-span-2 space-y-4">
   499→            <div className="relative">
   500→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   501→              <Input
   502→                placeholder="Rechercher par nom, DCI ou code-barres..."
   503→                className="pl-9 h-11"
   504→                value={search}
   505→                onChange={e => setSearch(e.target.value)}
   506→              />
   507→            </div>
   508→
   509→            <ScrollArea className="h-[calc(100vh-360px)]">
   510→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   511→                {filteredMeds.map(med => {
   512→                  const totalStock = med.lots.reduce((s, l) => s + l.quantite, 0)
   513→                  const inCart = cart.find(c => c.medicament.id === med.id)
   514→
   515→                  return (
   516→                    <Card
   517→                      key={med.id}
   518→                      className={`cursor-pointer transition-all hover:shadow-md ${inCart ? 'ring-2 ring-primary' : ''}`}
   519→                      onClick={() => addToCart(med)}
   520→                    >
   521→                      <CardContent className="p-3">
   522→                        <div className="flex justify-between items-start">
   523→                          <div className="flex-1 min-w-0">
