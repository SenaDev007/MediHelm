   127→   127→function formatFCFA(amount: number): string {
   128→   128→  return new Intl.NumberFormat('fr-FR', {
   129→   129→    style: 'decimal',

   135→   135→function formatDate(date: string | null): string {
   136→   136→  if (!date) return '—'
   137→   137→  return new Date(date).toLocaleDateString('fr-FR', {
   138→   138→    day: '2-digit',
   139→   139→    month: '2-digit',
   140→   140→    year: 'numeric',
   141→   141→  })
   142→   142→}
