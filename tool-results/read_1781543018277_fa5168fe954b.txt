   138→   138→function daysUntil(dateStr: string): number {
   139→   139→  const now = new Date()
   140→   140→  const target = new Date(dateStr)
   141→   141→  const diff = target.getTime() - now.getTime()
   142→   142→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
   143→   143→}

   145→   145→function getStockTotal(medicament: Medicament): number {
   146→   146→  return medicament.lots.reduce((sum, lot) => sum + lot.quantite, 0)

   149→   149→type StockStatus = 'OK' | 'ALERTE' | 'RUPTURE'
   150→   150→
   151→   151→function getStockStatus(medicament: Medicament): StockStatus {
   152→   152→  const total = getStockTotal(medicament)
   153→   153→  if (total === 0) return 'RUPTURE'
   154→   154→  if (total <= medicament.stockMinimum) return 'ALERTE'
   155→   155→  return 'OK'
   156→   156→}

   169→   169→function getDaysRemainingBadge(days: number) {
   170→   170→  if (days <= 0) {
   171→   171→    return <Badge variant="destructive" className="text-xs">Expiré</Badge>
   172→   172→  }
   173→   173→  if (days < 30) {
   174→   174→    return <Badge variant="destructive" className="text-xs">{days}j</Badge>
   175→   175→  }
   176→   176→  if (days < 60) {
   177→   177→    return <Badge className="bg-orange-100 text-orange-700 hover:bg-orange-100 border-0 text-xs">{days}j</Badge>
   178→   178→  }
   179→   179→  if (days <= 90) {
   180→   180→    return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100 border-0 text-xs">{days}j</Badge>
   181→   181→  }
   182→   182→  return <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-0 text-xs">{days}j</Badge>
   183→   183→}
