   195→      <Badge className={`text-[9px] ${info.className}`}>
   196→        <span className="mr-1">{getCanalIcon(canal)}</span>

   206→      TERMINEE: { label: 'Terminée', className: 'bg-[#E1F5EE] text-[#085041] border-0' },
   207→      ECHOUEE: { label: 'Échouée', className: 'bg-red-100 text-red-800 border-0' },

   225→      toast.error('Erreur lors de la mise à jour')

   233→        method: 'PATCH',

   242→      toast.error('Erreur lors de la mise à jour')

   253→      setSelectedNotif(null)

   255→      toast.error('Erreur lors de la suppression')

   258→

   261→      toast.error('Titre et message sont requis')

   265→      const res = await fetch('/api/notifications', {
   266→        method: 'POST',
   267→        headers: { 'Content-Type': 'application/json' },
   268→        body: JSON.stringify({
   269→          pharmacieId: pharmacie.id,

   277→        toast.success('Notification créée avec succès')
   278→        setAddNotifDialogOpen(false)

   282→        toast.error('Erreur lors de la création de la notification')

   285→      toast.error('Erreur lors de la création de la notification')

   296→        method: 'POST',
   297→        headers: { 'Content-Type': 'application/json' },

   299→          pharmacieId: pharmacie.id,
   300→          titre: campTitre,
   301→          message: campMessage,
   302→          destinataires: campDestinataires,
   303→          dateEnvoi: campDateEnvoi === 'now' ? new Date().toISOString() : new Date(campDateEnvoi).toISOString(),

   305→      })

   307→        toast.success('Campagne créée avec succès')
   308→        setCampagneDialogOpen(false)
   309→        resetCampagneForm()
   310→        fetchAllData()

   312→        toast.error('Erreur lors de la création de la campagne')

   315→      toast.error('Erreur lors de la création de la campagne')
   316→    }

   332→      setCampDestinataires(patients.map(p => p.id))

   383→              <Send className="w-8 h-8 text-[#1D9E75]/30" />

   392→                <span className="text-xl font-bold block text-blue-600">{pushCount}</span>

   394→              <Radio className="w-8 h-8 text-blue-400/30" />

   403→                <span className="text-xl font-bold block text-green-600">{smsCount}</span>

   430→              <Input

   444→                </Button>
   445→              </div>
   446→            )}
   447→            {activeTab === 'campagnes' && (
   448→              <Button className="gap-1 bg-[#1D9E75] hover:bg-[#085041]" size="sm" onClick={() => { resetCampagneForm(); setCampagneDialogOpen(true) }}>
   449→                <Plus className="w-3.5 h-3.5" /> Campagne
   450→              </Button>
   451→            )}
   452→          </div>
   453→        </div>
   454→
   455→        {/* Notifications Tab */}
   456→        <TabsContent value="notifications" className="mt-4">
   457→          {/* Filters */}
   458→          <Card className="mb-4">
   459→            <CardContent className="p-4">
   460→              <div className="flex flex-col sm:flex-row gap-3">
   461→                <Select value={filterCanal} onValueChange={setFilterCanal}>
   462→                  <SelectTrigger className="w-44">
   463→                    <SelectValue />
   464→                  </SelectTrigger>
   465→                  <SelectContent>
   466→                    <SelectItem value="all">Tous les canaux</SelectItem>
   467→                    <SelectItem value="PUSH">Push</SelectItem>
   468→                    <SelectItem value="SMS">SMS</SelectItem>
   469→                    <SelectItem value="EMAIL">Email</SelectItem>
   470→                    <SelectItem value="IN_APP">In-App</SelectItem>

   479→                    <SelectItem value="lu">Lus</SelectItem>
   480→                    <SelectItem value="non_lu">Non lus</SelectItem>
   481→                  </SelectContent>
   482→                </Select>
   483→                {(filterCanal !== 'all' || filterLu !== 'all') && (

   489→            </CardContent>

   491→
   492→          <Card>
   493→            <CardContent className="p-0">
   494→              <div className="overflow-x-auto">
   495→                <Table>
   496→                  <TableHeader>

   499→                      <TableHead>Date</TableHead>
   500→                      <TableHead>Titre</TableHead>
   501→                      <TableHead>Message</TableHead>
   502→                      <TableHead className="text-center">Canal</TableHead>

   521→                          <TableCell className="text-sm text-muted-foreground whitespace-nowrap">
