   115→// === Helpers ===
   116→
   117→function formatDate(dateStr: string): string {
   118→  try {
   119→    return new Date(dateStr).toLocaleDateString('fr-FR', {
   120→      day: '2-digit',
   121→      month: '2-digit',
   122→      year: 'numeric',
   123→    })
   124→  } catch {
   125→    return dateStr
   126→  }
   127→}
   128→
   129→function statutOrdonnanceLabel(statut: StatutOrdonnance): string {
   130→  const labels: Record<StatutOrdonnance, string> = {
   131→    RECUE: 'Reçue',
   132→    EN_VERIFICATION: 'En vérification',
   133→    VALIDEE: 'Validée',
   134→    PARTIELLEMENT_DELIVREE: 'Partiellement délivrée',
   135→    DELIVREE: 'Délivrée',
   136→    REFUSEE: 'Refusée',
   137→  }
   138→  return labels[statut] || statut
   139→}
   140→
   141→function statutOrdonnanceVariant(statut: StatutOrdonnance): 'default' | 'secondary' | 'destructive' | 'outline' {
   142→  switch (statut) {
   143→    case 'RECUE':
   144→      return 'secondary'
   145→    case 'EN_VERIFICATION':
   146→      return 'outline'
   147→    case 'VALIDEE':
   148→      return 'default'
   149→    case 'PARTIELLEMENT_DELIVREE':
   150→      return 'outline'
   151→    case 'DELIVREE':
   152→      return 'default'
   153→    case 'REFUSEE':
   154→      return 'destructive'
   155→    default:
   156→      return 'secondary'
   157→  }
   158→}
   159→
   160→function statutOrdonnanceBadge(statut: StatutOrdonnance) {
   161→  const colorMap: Record<StatutOrdonnance, string> = {
   162→    RECUE: 'bg-slate-100 text-slate-700 border-slate-200',
   163→    EN_VERIFICATION: 'bg-amber-50 text-amber-700 border-amber-200',
   164→    VALIDEE: 'bg-emerald-50 text-emerald-700 border-emerald-200',
   165→    PARTIELLEMENT_DELIVREE: 'bg-orange-50 text-orange-700 border-orange-200',
   166→    DELIVREE: 'bg-green-50 text-green-700 border-green-200',
   167→    REFUSEE: 'bg-red-50 text-red-700 border-red-200',
   168→  }
   169→  return (
   170→    <Badge variant={statutOrdonnanceVariant(statut)} className={`${colorMap[statut]} border`}>
   171→      {statutOrdonnanceLabel(statut)}
   172→    </Badge>
   173→  )
   174→}
