    34→    34→1. Architecture v2.0 — Nouveaux Composants Institutionnels
    35→    35→1.1 Vue d'ensemble enrichie
    36→    36→La v2.0 ajoute une couche institutionnelle à l'architecture existante. Trois nouveaux modules NestJS (dpmed-gateway, grossistes-gateway, sobaps-gateway) gèrent les intégrations avec les acteurs de la chaîne pharmaceutique. Un espace frontend dédié (portail-institutionnel) est servi sur un sous-domaine séparé. Une queue BullMQ haute priorité gère la diffusion des alertes sanitaires.

    82→    82→13. Tests v2.0 — Parcours Institutionnels
    83→    83→13.1 Nouveaux parcours E2E critiques
    84→    84→Alerte DPMED complète : envoi webhook DPMED > validation signature > identification pharmacies et patients concernés > diffusion push + SMS < 2 min > acquittement pharmacien > mise à jour portail DPMED.
    85→    85→Commande API grossiste : pharmacien valide commande > transmise API UbiPharm > confirmation webhook UbiPharm > mise à jour statut dans MédiHelm > notification pharmacien.
    86→    86→Signalement EI : pharmacien remplit formulaire > soumission API DPMED > accusé réception > référence DPMED enregistrée > mise à jour score conformité.
    87→    87→Réception SoBAPS avec écart : scan bon livraison > saisie quantités réelles ≠ quantités BL > écart détecté > stock mis à jour du reçu > webhook écart envoyé à SoBAPS.
    88→    88→Score conformité : vérification que le calcul des 5 composantes est correct sur données de test connues.
    89→    89→Portail DPMED : connexion rôle DPMED_ADMIN > émission alerte > consultation taux acquittement temps réel.
    90→    90→Sécurité webhook : envoi webhook DPMED avec signature invalide > rejet 403 > aucune alerte créée.

   472→   472→  const interactions: Interaction[] = [];

   474→   474→  for (let i = 0; i < dcis.length; i++) {

   482→   482→      const inter = fiches.find(x =>
   483→   483→        x.dci.toLowerCase().includes(dcis[j].toLowerCase())
   484→   484→      );
   485→   485→
   486→   486→      if (inter && (inter.gravite === 'MAJEURE' || inter.gravite === 'MODEREE')) {
   487→   487→        interactions.push({
   488→   488→          dci1: dcis[i], dci2: dcis[j],
   489→   489→          gravite: inter.gravite,
   490→   490→          description: inter.description
   491→   491→        });
   492→   492→      }
   493→   493→    }
   494→   494→  }
   495→   495→  return interactions; // vide = pas d'interaction connue

   605→   605→async diffuserAlerteComplete(alerteId: string) {
   606→   606→  const alerte = await this.prisma.alerteDPMED.findUniqueOrThrow({
   607→   607→    where: { id: alerteId }
   608→   608→  });
   609→   609→
   610→   610→  // 1. Vérifier la signature numérique de l'alerte
   611→   611→  const valid = await this.verifierSignature(alerte.signatureNumerique, alerte);
   612→   612→  if (!valid) throw new Error('MH-SEC-001: Signature alerte invalide — diffusion annulée');
   613→   613→
   614→   614→  // 2. Identifier les pharmacies concernées
   615→   615→  //    (ayant le lot concerné en stock actif)
   616→   616→  const pharmaciesConcernees = await this.prisma.$queryRaw`
   617→   617→    SELECT DISTINCT l.pharmacie_id, array_agg(l.numero_lot) as lots
   618→   618→    FROM lots l
   619→   619→    JOIN medicaments m ON l.medicament_id = m.id
   620→   620→    WHERE l.quantite > 0
   621→   621→    AND m.dci ILIKE ${alerte.dciConcernee}
   622→   622→    AND l.numero_lot = ANY(${alerte.numerosLotConcernes})
   623→   623→    GROUP BY l.pharmacie_id
   624→   624→  `;
   625→   625→
   626→   626→  // 3. Identifier les patients concernés
   627→   627→  //    (ayant acheté le lot dans les 90 derniers jours)
   628→   628→  const patientsConcernes = await this.getPatientsConcernes(alerte, 90);
   629→   629→
   630→   630→  // 4. Diffusion multi-canal aux pharmaciens
   631→   631→  for (const pharmacie of pharmaciesConcernees) {
   632→   632→    await this.prisma.diffusionAlerte.create({
   633→   633→      data: { alerteId, pharmacieId: pharmacie.pharmacie_id,
   634→   634→              lotsConcernes: pharmacie.lots,
   635→   635→              canalEnvoi: ['PUSH', 'SMS'] }
   636→   636→    });
   637→   637→    await this.notifService.pushAlerteDPMED(pharmacie.pharmacie_id, alerte);
   638→   638→    await this.smsService.sendAlerteDPMED(pharmacie.pharmacie_id, alerte);
   639→   639→  }
   640→   640→
   641→   641→  // 5. Diffusion aux patients concernés
   642→   642→  for (const patient of patientsConcernes) {
   643→   643→    await this.notifService.pushAlertePatient(patient, alerte);
   644→   644→    if (patient.telephone) await this.smsService.sendAlertePatient(patient, alerte);
   645→   645→  }
   646→   646→
   647→   647→  // 6. Mise à jour compteurs
   648→   648→  await this.prisma.alerteDPMED.update({
   649→   649→    where: { id: alerteId },
   650→   650→    data: {
   651→   651→      statut: 'DIFFUSEE',
   652→   652→      nbOfficinesNotifiees: pharmaciesConcernees.length,
   653→   653→      nbPatientsNotifies: patientsConcernes.length
   654→   654→    }
   655→   655→  });
   656→   656→}

   682→   682→async calculerScoreConformite(pharmacieId: string): Promise<ScoreConformite> {
   683→   683→
   684→   684→  // 1. Registre stupéfiants (25 pts)
   685→   685→  //    Vérifie que toutes les délivrances de stupéfiants ont un numéro de registre
   686→   686→  const stupMissing = await this.prisma.vente.count({
   687→   687→    where: { pharmacieId,
   688→   688→             lignes: { some: { medicament: { estStupefiant: true } } },
   689→   689→             ordonnanceId: null  // sans ordonnance liée
   690→   690→           }
   691→   691→  });
   692→   692→  const scoreStup = stupMissing === 0 ? 25 : Math.max(0, 25 - stupMissing * 5);
   693→   693→
   694→   694→  // 2. Alertes DPMED traitées < 24h (25 pts)
   695→   695→  const alertes = await this.prisma.diffusionAlerte.findMany({
   696→   696→    where: { pharmacieId, dateEnvoi: { gte: subDays(new Date(), 90) } }
   697→   697→  });
   698→   698→  const traitees24h = alertes.filter(a =>
   699→   699→    a.dateAcquittement &&
   700→   700→    differenceInHours(a.dateAcquittement, a.dateEnvoi) <= 24
   701→   701→  ).length;
   702→   702→  const scoreAlertes = alertes.length === 0 ? 25
   703→   703→    : Math.round((traitees24h / alertes.length) * 25);
   704→   704→
   705→   705→  // 3. Documents valides (20 pts)
   706→   706→  const docsExpires = await this.prisma.documentReglementaire.count({
   707→   707→    where: { pharmacieId, statut: { in: ['EXPIRE', 'EXPIRE_BIENTOT'] } }
   708→   708→  });
   709→   709→  const scoreDocs = docsExpires === 0 ? 20 : Math.max(0, 20 - docsExpires * 5);
   710→   710→
   711→   711→  // 4. Signalements EI soumis (15 pts)
   712→   712→  const eiNonSoumis = await this.prisma.signalementEI.count({
   713→   713→    where: { pharmacieId, statutEnvoi: 'EN_ATTENTE',
   714→   714→             createdAt: { lte: subDays(new Date(), 7) } }  // > 7j non soumis
   715→   715→  });
   716→   716→  const scoreEI = eiNonSoumis === 0 ? 15 : Math.max(0, 15 - eiNonSoumis * 5);
   717→   717→
   718→   718→  // 5. PV destructions à jour (15 pts)
   719→   719→  const destructSansVP = await this.prisma.destructionMedicament.count({
   720→   720→    where: { pharmacieId, pvUrl: null }
   721→   721→  });
   722→   722→  const scoreDestr = destructSansVP === 0 ? 15 : Math.max(0, 15 - destructSansVP * 5);
   723→   723→
   724→   724→  const scoreTotal = scoreStup + scoreAlertes + scoreDocs + scoreEI + scoreDestr;
   725→   725→
   726→   726→  return this.prisma.scoreConformite.upsert({
   727→   727→    where: { pharmacieId },
   728→   728→    create: { pharmacieId, scoreTotal, scoreRegistreStup: scoreStup,
   729→   729→              scoreAlerteDPMED: scoreAlertes, scoreDocuments: scoreDocs,

   782→   782→async verifyDPMEDSignature(payload: any, signature: string): Promise<boolean> {
   783→   783→  const publicKey = await this.keyStore.getDPMEDPublicKey();
   784→   784→  const verify = crypto.createVerify('SHA256');
   785→   785→  verify.update(JSON.stringify(payload));
   786→   786→  return verify.verify(publicKey, signature, 'base64');
   787→   787→}
