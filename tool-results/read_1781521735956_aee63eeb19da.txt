    73→function formatDate(dateStr: string) {
    74→  return new Date(dateStr).toLocaleDateString('fr-FR', { day: '2-digit', month: 'short', year: 'numeric' })
    75→}
    76→
    77→function formatFCFA(amount: number) {
    78→  return new Intl.NumberFormat('fr-FR').format(Math.round(amount)) + ' FCFA'
    79→}

    90→const statutFlow = ['RECUE', 'EN_COURS_VALIDATION', 'VALIDEE', 'DELIVREE']

    92→function getStatutProgress(statut: string): number {
    93→  const idx = statutFlow.indexOf(statut)
    94→  if (idx === -1) return 0
    95→  return ((idx + 1) / statutFlow.length) * 100
    96→}

    98→export default function OrdonnancesPage() {

   101→  const [patients, setPatients] = useState<Patient[]>([])

   104→  const [search, setSearch] = useState('')

   115→
   116→  // Validation dialog

   119→  const [validateCommentaire, setValidateCommentaire] = useState('')
   120→
   121→  // Upload dialog with drag & drop
   122→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   123→  const [uploadOrdoId, setUploadOrdoId] = useState<string | null>(null)
   124→  const { upload, uploading, progress, error: uploadError } = useUpload()

   126→  const fileInputRef = useRef<HTMLInputElement>(null)
   127→
   128→  // Create sale from ordonnance

   130→

   132→    if (!pharmacie?.id) return
   133→    const res = await fetch(`/api/ordonnances?pharmacieId=${pharmacie.id}`)
   134→    if (res.ok) setOrdonnances(await res.json())
   135→  }
   136→
   137→  useEffect(() => {

   139→      Promise.all([

   141→        fetch(`/api/patients?pharmacieId=${pharmacie.id}`).then(r => r.ok ? r.json() : []),
   142→        fetch(`/api/medicaments?pharmacieId=${pharmacie.id}`).then(r => r.ok ? r.json() : []),
   143→      ]).then(([ordos, pats, meds]) => {

   145→        setPatients(pats)
   146→        setMedicaments(meds)
   147→        setLoading(false)
   148→      }).catch(() => { setLoading(false) })
   149→    }
   150→  }, [pharmacie?.id])

   153→    if (!search) return ordonnances
   154→    const q = search.toLowerCase()
   155→    return ordonnances.filter(o =>
   156→      o.prescripteurNom.toLowerCase().includes(q) ||
   157→      (o.patient && `${o.patient.nom} ${o.patient.prenom}`.toLowerCase().includes(q)) ||
   158→      o.numeroRegistre?.includes(q)
   159→    )
   160→  }, [ordonnances, search])
   161→
   162→  const ordoStupfiants = useMemo(() => ordonnances.filter(o => o.estStupefiant), [ordonnances])

   166→    if (!pharmacie?.id || !formPrescripteur) return
   167→    try {

   170→        posologie: l.posologie,
   171→        duree: l.duree,
   172→        quantite: parseInt(l.quantite) || 1,
   173→      }))
   174→
   175→      const hasStupfiant = lignes.some(l => {
   176→        const med = medicaments.find(m => m.id === l.medicamentId)
   177→        return med?.estStupefiant
   178→      })
   179→
   180→      const res = await fetch('/api/ordonnances', {
   181→        method: 'POST',
   182→        headers: { 'Content-Type': 'application/json' },
   183→        body: JSON.stringify({
   184→          pharmacieId: pharmacie.id,
   185→          patientId: formPatientId || null,
   186→          prescripteurNom: formPrescripteur,
   187→          prescripteurInami: formPrescripteurInami || null,
   188→          dateOrdonnance: formDate,
   189→          dateReception: new Date().toISOString(),
   190→          statut: 'RECUE',
   191→          estStupefiant: hasStupfiant,
   192→          lignes,
   193→        }),
   194→      })
   195→
   196→      if (res.ok) {
   197→        toast.success('Ordonnance créée')
   198→        setCreateOpen(false)
   199→        setFormPatientId('')
   200→        setFormPrescripteur('')
   201→        setFormPrescripteurInami('')
   202→        setFormDate(new Date().toISOString().split('T')[0])
   203→        setFormLignes([])
   204→        refreshOrdonnances()

   206→        toast.error("Erreur lors de la création")
   207→      }
   208→    } catch {
   209→      toast.error("Erreur lors de la création")
   210→    }
   211→  }
   212→
   213→  // Validate/Refuse/Start validation

   215→    if (!selected) return
   216→    try {
   217→      const res = await fetch(`/api/ordonnances/${selected.id}/valider`, {
   218→        method: 'POST',
   219→        headers: { 'Content-Type': 'application/json' },
   220→        body: JSON.stringify({
   221→          typeValidation: validateType,
   222→          commentaire: validateCommentaire || null,
   223→          utilisateurId: 'demo-admin',
   224→        }),

   228→          VALIDATION: 'Ordonnance validée',
   229→          REFUS: 'Ordonnance refusée',
   230→          EN_COURS: 'Validation en cours',

   232→        toast.success(labels[validateType] || 'Statut mis à jour')
   233→        setValidateDialogOpen(false)
   234→        setValidateCommentaire('')
   235→        refreshOrdonnances()
   236→        setSelected(null)
   237→      }
   238→    } catch {
   239→      toast.error('Erreur lors de la validation')
   240→    }
   241→  }

   246→      const res = await fetch(`/api/ordonnances/lignes/${ligneId}`, {
   247→        method: 'PATCH',
   248→        headers: { 'Content-Type': 'application/json' },
   249→        body: JSON.stringify({ delivree: !currentDelivree }),

   252→        toast.success(!currentDelivree ? 'Ligne marquée comme délivrée' : 'Ligne marquée comme non délivrée')

   258→          l.id === ligneId ? { ...l, delivree: !currentDelivree } : l

   262→    }
   263→  }
   264→
   265→  const handleDrag = useCallback((e: React.DragEvent) => {

   267→    e.stopPropagation()
   268→    if (e.type === 'dragenter' || e.type === 'dragover') {
   269→      setDragActive(true)
   270→    } else if (e.type === 'dragleave') {
   271→      setDragActive(false)
   272→    }

   281→          method: 'POST',
   282→          body: (() => {
   283→            const fd = new FormData()
   284→            fd.append('image', file)
   285→            return fd

   287→        })

   289→        setUploadDialogOpen(false)
   290→        setUploadOrdoId(null)
   291→        refreshOrdonnances()
   292→      }
   293→    } catch {
   294→      toast.error("Erreur lors du téléchargement")
   295→    }
   296→  }, [uploadOrdoId, upload, refreshOrdonnances])
   297→
   298→  const handleDrop = useCallback((e: React.DragEvent) => {
   299→    e.preventDefault()
   300→    e.stopPropagation()
   301→    setDragActive(false)
   302→    if (e.dataTransfer.files && e.dataTransfer.files[0]) {
   303→      doFileUpload(e.dataTransfer.files[0])
   304→    }
   305→  }, [doFileUpload])
   306→
   307→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   308→    if (e.target.files && e.target.files[0]) {
   309→      doFileUpload(e.target.files[0])
   310→    }

   312→

   317→      const lignes = selected.lignes

   320→          medicamentId: l.medicamentId,
   321→          quantite: l.quantite,
   322→          prixUnitaire: l.medicament.prixVente,
   323→          montant: l.medicament.prixVente * l.quantite,
   324→        }))

   326→      if (lignes.length === 0) {
   327→        toast.error('Toutes les lignes sont déjà délivrées')
   328→        return

   334→        method: 'POST',
   335→        headers: { 'Content-Type': 'application/json' },

   337→          pharmacieId: pharmacie.id,

   339→          patientId: selected.patient?.id || null,
   340→          ordonnanceId: selected.id,

   344→          montantPaye: montantTotal,
   345→          monnaieRendue: 0,
   346→          lignes,
   347→          paiements: [{ modePaiement: 'ESPECES', montant: montantTotal }],
   348→        }),
   349→      })
   350→
   351→      if (res.ok) {
   352→        toast.success('Vente créée à partir de l\'ordonnance')
   353→        setCreateSaleOpen(false)
   354→        refreshOrdonnances()

   357→      toast.error('Erreur lors de la création de la vente')

   360→

   362→    return <div className="space-y-4"><Skeleton className="h-8 w-48" /><Skeleton className="h-64" /></div>
   363→  }

   365→  return (

   367→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">

   369→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   370→            <FileText className="w-6 h-6 text-primary" />
   371→            Ordonnances
   372→          </h1>
   373→          <p className="text-sm text-muted-foreground">{ordonnances.length} ordonnances</p>
   374→        </div>
   375→        <Button className="gap-2" onClick={() => setCreateOpen(true)}>
   376→          <Plus className="w-4 h-4" /> Nouvelle ordonnance
   377→        </Button>
   378→      </div>

   380→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   381→        <TabsList>
