    79→// À la connexion réussie — redirection automatique selon le rôle
    80→function redirectApresConnexion(payload: JWTPayload): string {
    81→  switch (payload.tenantType) {
    82→    case 'PHARMACIE':     return 'https://app.medihelm.com/dashboard';
    83→    case 'GROSSISTE':     return 'https://grossiste.medihelm.com/dashboard';
    84→    case 'INSTITUTIONNEL':return 'https://institutionnel.medihelm.com/dashboard';
    85→    case 'PLATFORM_ADMIN':return 'https://admin.medihelm.com/dashboard';
    86→    default:              return 'https://medihelm.com/compte';
    87→  }
    88→}

   124→// apps/medihelm-api/src/main.ts
   125→async function bootstrap() {
   126→  const app = await NestFactory.create(AppModule);
   127→
   128→  // CORS — lire les origines depuis la variable d'env
   129→  const origins = process.env.CORS_ORIGINS?.split(',') ?? [];
   130→
   131→  app.enableCors({
   132→    origin: (origin, callback) => {
   133→      // Autoriser les requêtes sans origin (Postman, curl, SSR Next.js)
   134→      if (!origin) return callback(null, true);
   135→      if (origins.includes(origin)) return callback(null, true);
   136→      callback(new Error(`CORS bloqué : origine non autorisée — ${origin}`));
   137→    },
   138→    credentials: true,        // Autoriser les cookies et Authorization header
   139→    methods: ['GET','POST','PUT','PATCH','DELETE','OPTIONS'],
   140→    allowedHeaders: ['Content-Type','Authorization','X-Requested-With'],
   141→  });
   142→
   143→  // Préfixe global — tous les endpoints sous /v1
   144→  app.setGlobalPrefix('v1');
   145→
   146→  await app.listen(3000);
   147→}

   185→export function decodeJWT(token: string): JWTPayload | null {
   186→  try {
   187→    // Décodage BASE64 du payload (sans vérification de signature)
   188→    // La vérification de signature se fait côté API NestJS uniquement
   189→    const base64 = token.split('.')[1];
   190→    const decoded = JSON.parse(atob(base64));
   191→    return decoded as JWTPayload;
   192→  } catch {
   193→    return null;
   194→  }
   195→}
   196→
   197→export function isTokenExpired(token: string): boolean {
   198→  const payload = decodeJWT(token);
   199→  if (!payload) return true;
   200→  return payload.exp * 1000 < Date.now();
   201→}

   204→  if (typeof window === 'undefined') return null;
   205→  return localStorage.getItem(key);
   206→}

   208→export function setStoredToken(token: string, key = 'mh-token'): void {

   212→export function clearStoredToken(key = 'mh-token'): void {
   213→  localStorage.removeItem(key);

   216→// Redirection après login selon le rôle
   217→export function getRedirectUrl(payload: JWTPayload): string {
   218→  const appUrl          = process.env.NEXT_PUBLIC_APP_URL
   219→                       ?? 'https://app.medihelm.com';
   220→  const grossisteUrl    = process.env.NEXT_PUBLIC_GROSSISTE_URL
   221→                       ?? 'https://grossiste.medihelm.com';
   222→  const institutUrl     = process.env.NEXT_PUBLIC_INSTITUTIONNEL_URL
   223→                       ?? 'https://institutionnel.medihelm.com';
   224→  const adminUrl        = process.env.NEXT_PUBLIC_ADMIN_URL
   225→                       ?? 'https://admin.medihelm.com';
   226→
   227→  if (payload.role === 'PLATFORM_ADMIN')  return `${adminUrl}/dashboard`;
   228→  if (payload.tenantType === 'GROSSISTE') return `${grossisteUrl}/dashboard`;
   229→  if (['DPMED_ADMIN','SOBAPS_VIEWER','ABRP_VIEWER'].includes(payload.role))
   230→    return `${institutUrl}/dashboard`;
   231→  return `${appUrl}/dashboard`;
   232→}
